 \pgfmathsetmacro{\c}{\n/4}

 \draw[draw=\col,line width=.9pt,domain=-.12:.12,samples=41] plot ({\c+\x},{1.55*exp(-(\x/.045)^2)});
 \fill[\col] (\c,0) circle(1.6pt);
}
