\chapter{Theory of Sparse Regularization}
\chapteroverview{The theory of sparse recovery asks when a reconstruction is unique, how much noise changes its coefficients, and whether it identifies the correct nonzero locations. A small reconstruction error alone does not guarantee that these locations are recovered. We use convex geometry and dual certificates to establish recovery conditions for the Lasso and its constrained counterpart, then examine their implications for spike deconvolution on increasingly fine grids.}

\section{Existence and Uniqueness}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Existence}

We study the penalized problem~\eqref{eq-lasso-lagr-ip}, with $\lambda>0$, and its constrained counterpart~\eqref{eq-lasso-constr-ip}:
\eql{\label{eq-lasso-lagr}\tag{$\Pp_\la(y)$}
	\umin{x \in \RR^N} f_\la(x) \eqdef \frac{1}{2\la} \norm{y-Ax}^2 + \norm{x}_1
} 
and the limiting problem as $\la \rightarrow 0$,
\eql{\label{eq-lasso-constr}\tag{$\Pp_0(y)$}
	\umin{A x = y} \norm{x}_1
	= \umin{x} f_0(x) \eqdef \iota_{\Ll_y}(x) + \norm{x}_1.
} 
where $A \in \RR^{P \times N}$, and $\Ll_y \eqdef \enscond{x \in \RR^N}{Ax=y}$.

Recall that we observe noisy measurements
\eq{
	y=A x_0 + w
}
We seek conditions ensuring that $x_0$ solves $\Pp_0(Ax_0)$ in the noiseless case, and quantitative bounds on its distance to solutions of $\Pp_\lambda(Ax_0+w)$ when $\lambda$ is chosen according to the noise level.

The objective in~\eqref{eq-lasso-lagr} is continuous and coercive because $\norm{\cdot}_1$ is coercive, so a minimizer exists.
%
Uniqueness is not automatic: $A$ may have a nontrivial kernel, and $\norm{\cdot}_1$ is not strictly convex.
%
If $y \in \Im(A)$, the constraint set in~\eqref{eq-lasso-constr} is nonempty. A constrained minimizer also exists but need not be unique.

\subsection{Polytope Projection for the Constrained Problem}
\label{sec-polytope-proj}

The next proposition characterizes noiseless $\ell^1$ recovery geometrically.

\begin{prop}\label{prop-polytope-proj}
	We denote $B \eqdef \enscond{x \in \RR^N}{\norm{x}_1 \leq 1}$. For $x_0\neq0$, assuming $\Im(A)=\RR^P$,
	\eql{\label{eq-polytop-ident} 
		x_0 \text{ is a solution to } \Pp_0(Ax_0)
		\quad\Longleftrightarrow\quad
		A \frac{x_0}{\norm{x_0}_1} \in \partial (AB)
	} 
	where ``$\partial$'' denotes the boundary and $AB=\enscond{Ax}{x\in B}$.
\end{prop}

\begin{proof}
Set $r=\norm{x_0}_1>0$ and $q=Ax_0/r$. If $x_0$ is not optimal, there is $z$ with $Az=Ax_0$ and $\norm{z}_1<r$. Since $A$ is onto, $A^+$ is a bounded right inverse. For sufficiently small $h\in\RR^P$,
\eq{
 \norm{z/r+A^+h}_1\leq\norm{z}_1/r+\norm{A^+h}_1<1,
}
and $q+h=A(z/r+A^+h)\in AB$. Thus $q$ is an interior point of $AB$.

Conversely, if $q\neq0$ is interior, then $(1+t)q\in AB$ for some $t>0$. Choose $u\in B$ with $Au=(1+t)q$. The vector $z=ru/(1+t)$ satisfies $Az=Ax_0$ and $\norm{z}_1<r$, so $x_0$ is not optimal. If $q=0$, the zero vector is already a better feasible point. These two implications prove the equivalence.
\begin{figure}[tbp]
\centering
\BookDrawing{chapter-theory-of-sparse-regularization/polytope-proof/original/sparse-theory-polytope-proof.pdf}
\caption{Geometry of the polytope characterization of exact recovery.}
\label{fig-polytope-proof}
\end{figure}
\end{proof}

For a nonzero recovered vector $x_0$, its measurement $Ax_0$ lies on the boundary of the scaled polytope $\norm{x_0}_1 A B$. When $P$ is much smaller than $N$, projection can remove faces and prevent $\ell^1$ recovery. Chapter~\ref{chap-cs} studies random projections that preserve enough of the geometry of the $\ell^1$ ball in $\RR^N$ to allow recovery from $\RR^P$.

Identifiability is unchanged by positive scaling: if $x_0$ is identifiable, so is $\rho x_0$ for every $\rho>0$. More generally, the relevant face of $B$, and hence condition~\eqref{eq-polytop-ident}, depends only on $\sign(x_0)$.
%
For this geometric discussion, assume uniqueness and let $\Aa : y \mapsto x^\star$ map each datum $y$ to the solution of~\eqref{eq-lasso-constr}.
%
By~\eqref{eq-polytop-ident}, $A$ and $\Aa$ are inverse maps between the cones $\Cc_s = \enscond{x}{\sign(x)=s}$ and $A\Cc_s$ for recoverable sign patterns $s$. Including the lower-dimensional cones and the origin, their images $A\Cc_s$ partition $\RR^P$. Suppose that the columns $(a_j)_j$ of $A$ have unit norm. For $P=3$, intersecting the cones $A\Cc_s$ with the unit sphere in $\RR^3$ gives a spherical Delaunay subdivision, which is a triangulation under a general-position assumption; see Figure~\ref{fig-delaunay}. In higher dimensions, simplices replace triangles. The subdivision satisfies the empty-cap property: the spherical cap bounded by a triangle's circumcircle contains no signed column $\pm a_j$ in its interior.
%
Figure~\ref{fig-polytopes} illustrates these conclusions in $\RR^2$ and $\RR^3$.

\subsection{Optimality Conditions}

For an index set $I \subset \{1,\ldots,N\}$, write $A=(a_i)_{i=1}^N$ for the columns of $A$ and $A_I \eqdef (a_i)_{i \in I} \in \RR^{P \times |I|}$ for the corresponding submatrix. For a vector $x \in \RR^N$, write $x_I \eqdef (x_i)_{i \in I} \in \RR^{|I|}$ for its restriction.

The following proposition rephrases the first-order optimality conditions in a convenient form.

\begin{prop}\label{prop-first-order-lagr}
	$x_\la$ is a solution to~\eqref{eq-lasso-lagr} for $\la>0$ if and only if
	\eq{
		\eta_{\la,I} = \sign(x_{\la,I})
		\qandq
		\norm{\eta_{\la,I^c}}_\infty \leq 1
	}
	where we define
	\eql{\label{eq-defn-eta-lamb}
		I\eqdef \supp(x_\la) \eqdef \enscond{i}{ x_{\la,i} \neq 0 }, 
		\qandq	\eta_\la \eqdef \frac{1}{\la}A^*(y-Ax_\la).
	}
\end{prop}
\begin{proof}
The objective in~\eqref{eq-lasso-lagr} is the sum of a differentiable convex function and a continuous convex penalty. The subdifferential sum rule gives
\eq{
	\partial f_\la(x) = \frac{1}{\la}A^*(Ax-y)  + \partial\norm{\cdot}_1(x).
} 
Thus $x_\la$ is a solution to~\eqref{eq-lasso-lagr} if and only if $0 \in \partial f_\la(x_\la)$, which gives the desired result.
\end{proof}

In particular, $\supp(x_\la) \subset \sat(\eta_\la)$, where $\sat(\eta)=\{i:|\eta_i|=1\}$ is the saturation set.

For the constrained case $\la=0$, the next proposition introduces dual certificates arising from the Lagrange multipliers associated with $\Ll_y$.

\begin{prop}\label{prop-first-order-constr}
	$x^\star$ solves~\eqref{eq-lasso-constr} if and only if $Ax^\star=y$ and
	\eql{\label{eq-defn-dual-certif}
		\exists \eta \in \Dd_0(y,x^\star) \eqdef \Im(A^*) \cap \partial \norm{\cdot}_1(x^\star).
	}
\end{prop}
\begin{proof}
The norm penalty in~\eqref{eq-lasso-constr} is continuous, so the subdifferential sum rule gives
\eq{
	\partial f_0(x) = \partial \iota_{\Ll_y}(x) + \partial\norm{\cdot}_1(x).
}
If $x \in \Ll_y$, then $\partial \iota_{\Ll_y}(x)$ is the normal space to the affine set $\Ll_y$, i.e. $\ker(A)^\bot=\Im(A^*)$.
\end{proof}

Every valid certificate satisfies $\supp(x^\star) \subset \sat(\eta)$, for $\eta\in \Dd_0(y,x^\star)$.

Writing $I=\supp(x^\star)$, one thus has
\eq{
	\Dd_0(y,x^\star) = \enscond{ \eta = A^* p }{
		\eta_I = \sign(x^\star_I), \:
		\norm{\eta}_\infty \leq 1
	}.
}
Although the notation $\Dd_0(y,x^\star)$ involves a particular solution $x^\star$, the set of certificates is the same for every primal solution. The duality argument below explains this independence.

\subsection{Uniqueness}

The next proposition shows that at least one minimizer uses linearly independent columns of $A$. It does not assert that every minimizer has this property.

\begin{prop}
	For $\lambda>0$, there is a solution $x^\star$ to~\eqref{eq-lasso-lagr} whose support $I$ satisfies $\ker(A_I)=\{0\}$. The same holds for~\eqref{eq-lasso-constr} when it is feasible.
\end{prop}

\begin{proof}
Let $x$ be a minimizer and $I=\supp(x)$. If $A_I$ is not injective, choose a nonzero $h_I\in\ker(A_I)$ and extend it by zero outside $I$. Along the interval containing $t=0$ on which $x+th$ retains the sign of $x$, the fidelity is constant and the $\ell^1$ norm is affine. Optimality at the interior point $t=0$ forces its slope to vanish. Move to a finite endpoint of this interval, where at least one coefficient becomes zero. Continuity shows that the endpoint is still a minimizer and has smaller support. Repeating the argument terminates after finitely many steps at a minimizer with injective $A_I$.
\end{proof}

\begin{figure}[tbp]
\centering
\BookDrawing{chapter-theory-of-sparse-regularization/sparse-theory-injectivity/original/sparse-theory-injectivity.pdf}
\caption{Coefficient trajectories along a support-reducing direction.}
\label{fig-review-sparse-theory-injectivity}
\end{figure}
%
If the columns of $A_I$ on a minimizer's support are linearly dependent, the proof constructs another minimizer. Thus that solution of~\eqref{eq-lasso-lagr} is nonunique.

For a solution $x_\la$ with $\ker(A_I)=\{0\}$, the optimality condition for~\eqref{eq-lasso-lagr} gives the implicit formula
\eql{\label{eq-local-shape-lasso}
	x_{\la,I} = A_I^+ y - \la (A_I^*A_I)^{-1} \sign(x_{\la,I}).
}
This expression generalizes soft thresholding (which is recovered when $A=\Id_N$).

Although the coefficient vector $x_\la$ may not be unique, its fitted data $Ax_\la$, also called the predictor, are unique.
%
Multiplying~\eqref{eq-local-shape-lasso} by the restricted measurement matrix gives
\eql{\label{eq-local-predictor-lasso}
	A x_{\la} = \Proj_{\Im(A_I)} y - \la A_I (A_I^*A_I)^{-1} \sign(x_{\la,I}).
}
Thus, up to an $O(\la)$ bias, this predictor is an orthogonal projection onto a low-dimensional subspace indexed by $I$.


\begin{lem}\label{lem-unique-predict}
	For $\lambda>0$, if $x_1,x_2$ solve~\eqref{eq-lasso-lagr}, then $Ax_1=Ax_2$. For the feasible constrained problem~\eqref{eq-lasso-constr}, both predictors equal $y$.
\end{lem}
\begin{proof}
    For $\la=0$, feasibility gives $Ax_1=Ax_2=y$. Otherwise, suppose $Ax_1 \neq Ax_2$ and set $x=(x_1+x_2)/2$. Convexity gives
	\eq{
		\norm{x}_1 \leq \frac{ \norm{x_1}_1 + \norm{x_2}_1 }{2}
		\qandq
		\norm{Ax-y}^2 < \frac{ \norm{Ax_1-y}^2 + \norm{Ax_2-y}^2 }{2}		
	}
    where strict convexity of the squared Euclidean norm makes the second inequality strict. Hence
	\eq{
		\frac{1}{2\la}\norm{Ax-y}^2 +\norm{x}_1 <
		\frac{1}{2\la}\norm{Ax_1-y}^2 +\norm{x_1}_1, 
	}
    contradicting the optimality of $x_1$.
\end{proof}


\begin{prop}\label{sec-prop-uniquness-lagr}
 	For $\la>0$, let $x_\la$ be a solution to~\eqref{eq-lasso-lagr} and denote $\eta_\la \eqdef \frac{1}{\la}A^*(y-A x_\la)$.
	%
	We define the ``extended support'' as 
	\eq{
		J \eqdef \sat(\eta_\la) \eqdef \enscond{i}{|\eta_{\la,i}|=1}.
	}
	If $\ker(A_{J})=\{0\}$ then $x_\la$ is the unique solution of~\eqref{eq-lasso-lagr}.
\end{prop}
\begin{proof}
	If $\tilde x_\la$ is also a minimizer, then by Lemma~\ref{lem-unique-predict}, 
	$Ax_\la=A\tilde x_\la$, so that in particular they share the same dual certificate
	\eq{
		\eta_\la = \frac{1}{\la}A^*(y-Ax_\la) = \frac{1}{\la}A^*(y-A\tilde x_\la).
	}
	The first-order condition, Proposition~\ref{prop-first-order-lagr}, shows that necessarily
	$\supp(x_\la) \subset J$ and $\supp(\tilde x_\la) \subset J$.
	%
	Since $A_J x_{\la,J} = A_J \tilde x_{\la,J}$, and since $\ker(A_J)=\{0\}$, one has 
	$x_{\la,J}=\tilde x_{\la,J}$, and thus $x_\la=\tilde x_\la$ because their supports are included in $J$.
\end{proof}

\begin{prop}\label{sec-prop-uniquness-constr}
 	Let $x^\star$ be a solution to~\eqref{eq-lasso-constr}. If there exists $\eta \in \Dd_0(y,x^\star)$ such that 
	$\ker(A_{J})=\{0\}$ where $J \eqdef \sat(\eta)$ then $x^\star$ is the unique solution of~\eqref{eq-lasso-constr}.
\end{prop}
\begin{proof}
	The proof is the same as for Proposition~\ref{sec-prop-uniquness-lagr}, replacing $\eta_\la$ by $\eta$. 
\end{proof}

For $\lambda>0$, a joint density for the entries of $A$ on $\RR^{P\times N}$ implies that its columns are in general position almost surely. The Lasso solution is then unique for every $y$. Randomizing only $y$, while keeping $A$ fixed, does not in general ensure uniqueness.

\subsection{Duality}
\label{sec-duality-lasso}

The first-order conditions can also be derived from the convex duality theory in Section~\ref{sec-cvx-duality}. This interpretation explains why certificates characterize optimal recovery.


\begin{thm}
	For $\lambda>0$, strong duality holds between~\eqref{eq-lasso-lagr} and the following dual problem. For $\lambda=0$, the same statement holds with~\eqref{eq-lasso-constr} as the primal problem, provided $y\in\Im(A)$:
	\eql{\label{eq-dual-lasso}
		 \usup{p \in \RR^P} \enscond{ \dotp{y}{p} - \frac{\la}{2} \norm{p}^2 }{ \norm{A^* p}_\infty \leq 1 }.
	}
	A pair $(x^\star,p^\star)$ is primal--dual optimal if and only if it satisfies the following certificate condition together with the residual relation below:
	\eql{\label{eq-pd-support-localization}
		A^* p^\star \in \partial \norm{\cdot}_1(x^\star)
		\quad\Leftrightarrow\quad
		\pa{
        \norm{A^*p^\star}_\infty\leq1,\quad
		I \subset \sat(A^* p^\star)
		\qandq
		\sign(x_I^\star) = A_I^* p^\star
		}, 
	}
	where we denote $I=\supp(x^\star)$,
	%
	and furthermore, for $\la>0$, 
	\eq{
		p^\star = \frac{y-A x^\star}{\la}.
	}
	while for $\la=0$, $Ax^\star=y$.
\end{thm}

\begin{proof}
Introduce $z=Ax$ and define $g_\lambda(z)=\norm{z-y}^2/(2\lambda)$ when $\lambda>0$, and $g_0=\iota_{\{y\}}$. The Lagrangian is
\eq{
 \Ll(x,z,p)=g_\lambda(z)+\norm{x}_1+\langle z-Ax,p\rangle.
}
For $\lambda>0$, continuity gives strong duality. For $\lambda=0$, feasibility and polyhedral convex duality give the same conclusion. Minimizing the Lagrangian over $x$ and $z$ yields
\eq{
 \sup_{p\in\RR^P}\big[-g_\lambda^*(-p)-(\norm{\cdot}_1)^*(A^*p)\big].
}
The conjugates are
\eq{
 g_\lambda^*(q)=\langle y,q\rangle+\frac\lambda2\norm q^2,
 \qquad (\norm{\cdot}_1)^*=\iota_{\{\norm{\cdot}_\infty\leq1\}},
}
including $\lambda=0$, which gives~\eqref{eq-dual-lasso}. This is also a direct application of the Fenchel--Rockafellar formula~\eqref{eq-fenchel-Rockafellar}.

The optimality conditions are $z^\star=Ax^\star$, $A^*p^\star\in\partial\norm{\cdot}_1(x^\star)$, and $-p^\star\in\partial g_\lambda(z^\star)$. For $\lambda>0$, the last condition gives $p^\star=(y-Ax^\star)/\lambda$. For $\lambda=0$, it is equivalent to $Ax^\star=y$. The first condition on $p^\star$ gives the support and sign relations in~\eqref{eq-pd-support-localization}.
\end{proof}


For $\la>0$, the dual objective~\eqref{eq-dual-lasso} is strongly concave. Completing the square shows that its maximizer $p_\la$ is an orthogonal projection:
\eq{
	p_\la \in \uargmin{p \in \RR^P} \enscond{ \norm{p-y/\la} }{ \norm{A^* p}_\infty \leq 1 }. 
}
Thus the supremum is attained at a unique point for $\lambda>0$. For $\lambda=0$ and feasible data, a dual optimum also exists, though it need not be unique.

\subsection{Bregman Divergence Rates for General Regularizations}

Consider the more general regularized problem
\eql{\label{eq-lagr-J}
	\umin{x \in \RR^N} \frac{1}{2\la}\norm{Ax-y}^2+J(x)
}
for a proper convex regularizer $J$ and $\lambda>0$. We state estimates for any minimizer, without assuming uniqueness.

\begin{figure}[tbp]
\centering
\BookDrawing{chapter-theory-of-sparse-regularization/bregman/original/sparse-theory-bregman.pdf}
\caption{Visualization of Bregman divergences.}
\label{fig-bregman}
\end{figure}

For $x_0$ with nonempty subdifferential and a chosen $\eta \in \partial J(x_0)$, define the Bregman divergence
\eq{
	D_\eta(x|x_0) \eqdef J(x)-J(x_0)-\dotp{\eta}{x-x_0}.
}
One has $D_\eta(x_0|x_0)=0$ and, by convexity, $D_\eta(x|x_0)\geq0$; see Figure~\ref{fig-bregman}.

If $J$ is differentiable, then $\partial J(x_0)=\{\nabla J(x_0)\}$, and the divergence becomes
\eq{
	D(x|x_0) \eqdef J(x)-J(x_0)-\dotp{\nabla J(x_0)}{x-x_0}.
}
If $J$ is strictly convex, then $D(x|x_0)=0$ exactly when $x=x_0$. Thus $D(\cdot|\cdot)$ separates points, although it need not be symmetric or satisfy the triangle inequality.

If $J=\norm{\cdot}^2$, then $D(x|x_0) = \norm{x-x_0}^2$ is the squared Euclidean distance.
%
For the negative-entropy functional $J(x)=\sum_i x_i(\log x_i-1)+\iota_{\RR_+^N}(x)$, one obtains
\eq{
	D(x|x_0)  = \sum_i x_i \log\pa{ \frac{x_i}{x_{0,i}} } + x_{0,i}-x_i
}
This is the generalized Kullback--Leibler divergence for $x_0\in\RR_{++}^N$ and $x\in\RR_+^N$, with the convention $0\log0=0$. For probability vectors, the linear terms sum to zero.

The following estimate, associated with the work of Burger and Osher, yields a linear noise rate measured by Bregman divergence.

\begin{thm}\label{thm-bregman-rates}
	If there exists 
	\eql{\label{eq-sourcecond-J}
		\eta  = A^* p \in \Im(A^*) \cap \partial J(x_0), 
	}
    then every solution $x_\la$ of~\eqref{eq-lagr-J} satisfies
	\eql{\label{eq-bregman-rate}
		D_\eta( x_\la|x_0 ) \leq \frac{1}{2}\pa{ \frac{\norm{w}}{\sqrt{\la}} + \sqrt{\la}\norm{p} }^2.
	}
    The measurement residual also satisfies
	\eql{\label{eq-predict-source}
		\norm{Ax_\la-y} \leq \norm{w} + 2\norm{p}\la.
	}
\end{thm}
\begin{proof}
	The optimality of $x_\la$ for~\eqref{eq-lagr-J} implies
	\eq{
		\frac{1}{2\la}\norm{Ax_\la-y}^2+J(x_\la) \leq \frac{1}{2\la}\norm{Ax_0-y}^2+J(x_0) = \frac{1}{2\la}\norm{w}^2+J(x_0).
	}
	Hence, using $\dotp{\eta}{x_\la-x_0} = \dotp{p}{Ax_\la-Ax_0} = \dotp{p}{A x_\la - y+w}$, one has
	\begin{align*}
		D_\eta(x_\la|x_0) &= J(x_\la)-J(x_0)-\dotp{\eta}{x_\la-x_0} 
			\leq \frac{1}{2\la}\norm{w}^2 - \frac{1}{2\la}\norm{Ax_\la-y}^2 -\dotp{p}{A x_\la-y} - \dotp{p}{w} \\
			& = \frac{1}{2\la}\norm{w}^2 - \frac{1}{2\la}\norm{Ax_\la-y + \la p }^2 + \frac{\la}{2} \norm{p}^2 - \dotp{p}{w} \\
			& \leq \frac{1}{2\la}\norm{w}^2 + \frac{\la}{2} \norm{p}^2 + \norm{p}\norm{w}
			= \frac{1}{2}\pa{ \frac{\norm{w}}{\sqrt{\la}} + \sqrt{\la}\norm{p} }^2.
	\end{align*}
Since $D_\eta(x_\lambda|x_0)\geq0$, the completed-square bound gives
\eq{
 \norm{Ax_\lambda-y+\lambda p}^2
 \leq\norm{w-\lambda p}^2
 \leq(\norm{w}+\lambda\norm{p})^2.
}
The triangle inequality therefore yields
\eq{
 \norm{Ax_\lambda-y}\leq\norm{w}+2\lambda\norm{p}.
}

\end{proof}

When $\norm{w}>0$ and $p\neq0$, choose $\la=\norm{w}/\norm{p}$ in~\eqref{eq-bregman-rate}. The resulting estimate $D_\eta( x_\la|x_0 ) \leq 2 \norm{w}\norm{p}$ is linear in the noise level.
%
For the simple case of a quadratic regularizer $J(x)=\norm{x}^2/2$, as used in Section~\ref{sec-tikhonov}, the source condition~\eqref{eq-sourcecond-J} becomes
\eq{
	x_0 \in \Im(A^*)
}
This is~\eqref{eq-source-cond-init} with $\be=1$. Under this condition, \eqref{eq-bregman-rate} gives the sublinear $\ell^2$ estimate
\eq{
	\norm{x_0-x_\la} \leq 2 \sqrt{ \norm{w}\norm{p} }.
}
This agrees with Theorem~\ref{thm-sublin-quad} under the convention $x_0\in\Im((A^*A)^{\beta/2})$.

%
The source condition~\eqref{eq-sourcecond-J} is sufficient for $x_0$ to minimize $J$ subject to $Ax=Ax_0$. In finite dimension, it is also necessary under a subdifferential sum-rule qualification, for example when $J$ is finite and continuous on $\RR^N$. Without such a qualification, existence of a Lagrange multiplier is an additional assumption.

\subsection{Linear Rates in Norms for \texorpdfstring{$\ell^1$}{l1} Regularization}

For a regularizer $J$ that is not strictly convex, the Bregman bound~\eqref{eq-bregman-rate} need not control the error in norm. We now examine how additional certificate conditions restore such control for the $\ell^1$ penalty $J=\norm{\cdot}_1$.

    \begin{figure}[tbp]
\centering
\BookDrawing{chapter-theory-of-sparse-regularization/sparse-theory-bregman-l1/original/sparse-theory-bregman-l1.pdf}
\caption{Bregman divergence controls the $\ell^1$ error on coordinates where $\eta$ does not saturate.}
\label{fig-review-sparse-theory-bregman-l1}
\end{figure}
	
The next lemma quantifies this control through the gap between $|\eta_i|$ and one.

\begin{lem}
	For $\eta \in \partial \norm{\cdot}_1(x_0)$, let $J \eqdef \sat(\eta)$. Then
	\eql{\label{eq-bound-breg-source}
		D_\eta(x|x_0) \geq (1-\norm{\eta_{J^c}}_\infty) \norm{(x-x_0)_{J^c}}_1.
	}
\end{lem}

\begin{proof}
Note that $x_{0,J^c}=0$ since $\supp(x_0) \subset \sat(\eta)$ by definition of the subdifferential of the $\ell^1$ norm.
%
	Since the $\ell^1$ norm is separable, each term in the sum defining $D_\eta(x|x_0)$ is nonnegative, hence
	\begin{align*}
		D_\eta(x|x_0) &= \sum_i |x_i|-|x_{0,i}|- \eta_i ( x_{i}-x_{0,i} ) 
			\geq \sum_{i \in J^c} |x_i|-|x_{0,i}|- \eta_i ( x_{i}-x_{0,i} ) \\
		&= \sum_{i \in J^c} |x_i|- \eta_i x_{i} \geq \sum_{i \in J^c} (1-|\eta_i|) |x_i|
		\geq (1-\norm{\eta_{J^c}}_\infty) \sum_{i \in J^c} |x_i|
		= (1-\norm{\eta_{J^c}}_\infty) \sum_{i \in J^c} |x_i-x_{0,i}|.
	\end{align*}
\end{proof}

We use the convention $\norm{\eta_\emptyset}_\infty=0$. The margin $1-\norm{\eta_{J^c}}_\infty>0$ measures how far $\eta$ lies from saturation on the complementary coordinates. A larger margin gives stronger control of the coefficient error.

This lemma yields a norm convergence rate under the certificate and injectivity assumptions of Proposition~\ref{sec-prop-uniquness-constr}, with $x^\star=x_0$.

\begin{thm}\label{thm-linrate-l1}
	If there exists 
	\eql{\label{eq-sourcecond-l1}
		\eta \in \Dd_0(A x_0,x_0)
	}
	and $\ker(A_{J})=\{0\}$ where
	$J \eqdef \sat(\eta)$ then, for fixed $c>0$ and $\norm{w}>0$, choosing $\la=c\norm{w}$ gives a constant $C$ (depending on $A$, the certificate, and $c$, but not on $w$) such that any solution $x_\la$ of $\Pp_\la(Ax_0+w)$ satisfies
	\eql{\label{eq-linrate-l1}
		\norm{x_\la-x_0} \leq C \norm{w}. 
	}
\end{thm}
\begin{proof}
	Write $\eta=A^*p$ and $y=Ax_0+w$.
	%
	The optimality of $x_\la$ in~\eqref{eq-lasso-lagr} implies
	\eq{
		\frac{1}{2\la} \norm{Ax_\la-y}^2 + \norm{x_\la}_1 \leq 
		\frac{1}{2\la} \norm{Ax_0-y}^2 + \norm{x_0}_1 = \frac{1}{2\la} \norm{w}^2 + \norm{x_0}_1
	}
	and hence
	\eq{
		\norm{Ax_\la-y}^2 \leq \norm{w}^2 + 2\la \norm{x_0}_1
	}
	%
	Using the fact that $A_J$ is injective, one has $A_J^+A_J=\Id_{J}$, so that
	\begin{align*}
		\norm{(x_\la-x_0)_J}_1 &= \norm{A_J^+ A_J(x_\la-x_0)_J}_1
			\leq \norm{A_J^+}_{2,1} \norm{ A_J x_{\la,J} - y + w }
			\leq \norm{A_J^+}_{2,1} \pa{ \norm{ A_J x_{\la,J} - y } + \norm{w} }  \\
			& \leq 
			\norm{A_J^+}_{2,1} \pa{ \norm{ A x_{\la} - y } + \norm{A_{J^c} x_{\la,J^c}} + \norm{w} }\\
			&\leq 
			\norm{A_J^+}_{2,1} \pa{ \norm{ A x_{\la} - y } + \norm{A_{J^c}}_{1,2} \norm{ x_{\la,J^c} - x_{0,J^c} }_1 + \norm{w} } \\
			&\leq 
			\norm{A_J^+}_{2,1} \pa{ \norm{w} + 2\norm{p}\la + \norm{A_{J^c}}_{1,2} \norm{ x_{\la,J^c} - x_{0,J^c} }_1 + \norm{w} }
	\end{align*}
	where we used $x_{0,J^c}=0$ and~\eqref{eq-predict-source}. The mixed operator norm $\norm{B}_{p,q}$ maps $\ell^p$ to $\ell^q$, as defined in Remark~\ref{rem-operator-norm}.
	Substituting this bound into the decomposition and using~\eqref{eq-bound-breg-source} and~\eqref{eq-bregman-rate} gives
	\begin{align*}
		\norm{x_\la-x_0}_1 &= \norm{(x_\la-x_0)_J}_1 + \norm{(x_\la-x_0)_{J^c}}_1 \\
		& \leq \norm{(x_\la-x_0)_{J^c}}_1 \pa{ 1 + \norm{A_J^+}_{2,1} \norm{A_{J^c}}_{1,2} }
		+ \norm{A_J^+}_{2,1} \pa{   2\norm{p} \la + 2 \norm{w} }		\\
		& \leq \frac{D_\eta(x_\la|x_0)}{1-\norm{\eta_{J^c}}_\infty} \pa{ 1 + \norm{A_J^+}_{2,1} \norm{A_{J^c}}_{1,2} }
		+ \norm{A_J^+}_{2,1} \pa{   2\norm{p} \la + 2 \norm{w} } \\
		& \leq \frac{ 
				\frac{1}{2}\pa{ \frac{\norm{w}}{\sqrt{\la}} + \sqrt{\la}\norm{p} }^2
			}{1-\norm{\eta_{J^c}}_\infty} \pa{ 1 + \norm{A_J^+}_{2,1} \norm{A_{J^c}}_{1,2} }
		+ \norm{A_J^+}_{2,1} \pa{   2\norm{p} \la + 2 \norm{w} }.
	\end{align*}	
	Thus setting $\la=c\norm{w}$, one obtains the constant
	\eq{
		C \eqdef \frac{ 
				\frac{1}{2}\pa{ \frac{1}{\sqrt{c}} + \sqrt{c}\norm{p} }^2
			}{1-\norm{\eta_{J^c}}_\infty} \pa{ 1 + \norm{A_J^+}_{2,1} \norm{A_{J^c}}_{1,2} }
		+ \norm{A_J^+}_{2,1} \pa{   2\norm{p} c + 2  }.
	}
\end{proof}


This theorem guarantees uniqueness of the noiseless solution $x_0$, but does not imply uniqueness of $x_\la$.
%
The source condition~\eqref{eq-sourcecond-l1}, combined with the injectivity condition $\ker(A_{J})=\{0\}$, gives the $\ell^2$ stability estimate~\eqref{eq-linrate-l1}.
%
The estimate~\eqref{eq-linrate-l1} controls the error linearly relative to the fixed noiseless signal $x_0$. It does not by itself establish pairwise Lipschitz continuity between arbitrary noisy solutions.


Compare this linear rate with the sublinear rates for quadratic regularization in Theorem~\ref{thm-sublin-quad}.
%
The source conditions for linear regularization~\eqref{eq-source-cond-init} and nonlinear regularization~\eqref{eq-sourcecond-l1} express different structural assumptions.
%
In the quadratic case with $\be=1$, the source condition is $x_0\in\Im(A^*)=\ker(A)^\bot$ in finite dimension. The signal itself must have no component in $\ker(A)$ because squared-norm regularization cannot recover such a component.
%
The nonlinear condition instead requires a subgradient $\eta$ to belong to $\Im(A^*)$. It can therefore permit recovery of signal components in $\ker(A)$ when the prior supplies sufficient structure.

\subsection{Sparsistency for Low Noise}
\label{sec-sparsistency}

Theorem~\ref{thm-linrate-l1} gives an abstract guarantee whose certificate assumptions may be difficult to verify.
%
We now construct a candidate certificate explicitly. When it satisfies a strict off-support bound, it guarantees both a linear rate and support stability.


For any solution $x_\la$ of $(\Pp_\la(y))$, define the dual certificate as in~\eqref{eq-defn-eta-lamb}. Uniqueness of the fitted data makes it independent of the chosen solution:
\eq{
	\eta_\la \eqdef A^* p_\la
	\qwhereq
	p_\la \eqdef \frac{y - A x_\la}{\la}.
}
The next proposition shows that $p_\la$ converges to the dual solution of minimum norm for the constrained problem.

\begin{prop}
	If $y = A x_0$ where $x_0$ solves $\Pp_0(y)$, one has
	\eql{\label{eq-minnorm-certif}
		p_\la \rightarrow p_0 \eqdef
		\uargmin{p \in \RR^P} \enscond{\norm{p}}{ A^* p \in \Dd_0(y,x_0) }.
	}
	The vector $\eta_0\eqdef A^*p_0$ is called the minimum-norm certificate; the minimized norm is that of the data-space vector $p_0$.
\end{prop}
\begin{proof}
	Write $\mathcal D=\{p:\norm{A^*p}_\infty\leq1\}$ and let $M$ be the maximum of $\langle y,p\rangle$ on $\mathcal D$. Optimality relative to the minimum-norm dual maximizer $p_0$ gives $0\leq M-\langle y,p_\lambda\rangle\leq\frac\lambda2(\norm{p_0}^2-\norm{p_\lambda}^2)$. Thus $\norm{p_\lambda}\leq\norm{p_0}$, every cluster point is a dual maximizer of no greater norm, and uniqueness of $p_0$ proves convergence as $\lambda\downarrow0$.
\end{proof}

The certificates in $\Dd_0(y,x_0)$ for $\la=0$ may be nonunique, but the limit $\la\rightarrow 0$ selects one of them. This selected certificate governs support stability when both the noise and $\la$ are small.

The inequality constraint $\norm{\eta_0}_\infty\leq1$ makes the minimum-norm problem~\eqref{eq-minnorm-certif} difficult to solve explicitly. Dropping this $\ell^\infty$ constraint while retaining interpolation on the support gives the minimum-norm precertificate:
\eql{\label{eq-fuchs-precertif}
	\eta_F \eqdef A^* p_F
	\qwhereq 
	p_F \eqdef
	\uargmin{p \in \RR^P} \enscond{\norm{p}}{ A_I^* p = \sign(x_{0,I}) }.
}
The notation ``$\eta_F$'' refers to the ``Fuchs'' certificate, named after J.-J. Fuchs, who first used it to study $\ell^1$ minimization.

The vector $p_F$ may fail dual feasibility because $\norm{\eta_F}_\infty \leq 1$ need not hold. This is why the construction gives a precertificate.
%
If the interpolation system is consistent, $p_F$ is its minimum-norm solution: $A_I^*p=\sign(x_{0,I})$. The pseudoinverse gives $p_F=A_I^{*,+} \sign(x_{0,I})$; see Proposition~\ref{prop-pseudo-inv}. Under $\ker(A_I)=\{0\}$, this simplifies to
\eq{
	p_F = A_I (A_I^*A_I)^{-1} \sign(x_{0,I}).
}
With the Gram matrix $C \eqdef A^* A$, the certificate becomes
\eql{\label{eq-eta-f-cov}
	\eta_F = C_{\cdot,I} C_{I,I}^{-1} \sign(x_{0,I}).
}
The next proposition relates $\eta_F$ to $\eta_0$: whenever $\eta_F$ is feasible, it equals $\eta_0$.

\begin{prop}
	If $\norm{\eta_F}_\infty \leq 1$, then $p_F=p_0$ and $\eta_F=\eta_0$.
\end{prop}

The condition $\norm{\eta_F}_\infty \leq 1$ implies that $x_0$ is a solution to~\eqref{eq-lasso-constr}.
% 
A strict off-support bound on $\eta_F$ strengthens this recovery condition and gives both a linear error rate and support stability for small noise.

The feasibility condition $\norm{\eta_F}_\infty \leq 1$ has the empty-cap interpretation shown in Figure~\ref{fig-delaunay}. We thank Charles Dossal for pointing out this connection to spherical Delaunay triangulations.

\begin{rem}[Operator norm]\label{rem-operator-norm}
In the proof, we use the $\ell^p-\ell^q$ matrix operator norm, which is defined as
\eq{
	\norm{B}_{p,q} \eqdef \max \enscond{ \norm{B u}_q }{ \norm{u}_p \leq 1 }.
}
For $p=q$, we denote $\norm{B}_{p} \eqdef \norm{B}_{p,p}$.
%
For $p=q=2$, $\norm{B}_2$ is the maximum singular value, and one has
\eq{
	\norm{B}_1 = \umax{j} \sum_i |B_{i,j}|
	\qandq
	\norm{B}_\infty = \umax{i} \sum_j |B_{i,j}|.
}
\end{rem}

\begin{thm}\label{thm-support-stable}
Let $I=\supp(x_0)\neq\emptyset$, assume $A_I$ is injective, and suppose $\norm{\eta_{F,I^c}}_\infty<1$. There exist $c_0,c_1>0$, depending on $A$, $I$, and $x_0$, such that if
\eq{
 0<\lambda<c_0,\qquad \norm{A^*w}_\infty<c_1\lambda,
}
then the solution $x_\lambda$ of~\eqref{eq-lasso-lagr} is unique and has the same support and sign as $x_0$. It satisfies
\eql{\label{eq-explicit-lownoise}
 x_{\lambda,I}=x_{0,I}+A_I^+w-\lambda(A_I^*A_I)^{-1}\sign(x_{0,I}),
 \qquad x_{\lambda,I^c}=0.
}
In particular, the error is $O(\norm{A^*w}_\infty+\lambda)$. When $\norm{A^*w}_\infty>0$, choosing $\lambda$ proportional to this quantity with a sufficiently large constant gives an $O(\norm{w})$ error as the noise tends to zero.
\end{thm}

\begin{proof}
    Let $T \eqdef \min_{i \in I} |x_{0,i}|$ be the smallest nonzero signal amplitude, and let $\de \eqdef \norm{A^* w}_\infty$
    measure the noise after correlation with the measurement columns.
	%
    Set $s \eqdef \sign(x_0)$ and use~\eqref{eq-explicit-lownoise} to define the candidate
	\eql{
		\hat x_{I} \eqdef x_{0,I} + A_I^+ w - \la (A_I^*A_I)^{-1} s_I, 
	}
	and $\hat x_{I^c}=0$. The goal is to show that $\hat x$ is indeed the unique solution of~\eqref{eq-lasso-lagr}. 
	
	
    \noindent\textit{Step 1.} Sign consistency means $\sign(\hat x)=s$. It follows from $\norm{x_{0,I}-\hat x_I}_\infty < T$, for which a sufficient condition is
	\eql{\label{eq-supp-stab-proof-1}
		\norm{x_{0,I}-\hat x_I}_\infty \leq K \norm{A_I^* w}_\infty + K \la < T
		\qwhereq
		K \eqdef \norm{(A_I^*A_I)^{-1}}_{\infty}, 
	}
	where we used the fact that $A_I^+ = (A_I^*A_I)^{-1} A_I^*$.
	
    \noindent\textit{Step 2.} We verify the strict off-support bound in Proposition~\ref{sec-prop-uniquness-lagr}: $\norm{\hat \eta_{I^c}}_\infty < 1$, where $\la \hat \eta = A^* (y-A \hat x)$. Together with sign consistency and injectivity, this proves that $\hat x$ is the unique solution of~\eqref{eq-lasso-lagr}. Compute
	\begin{align*}
		\la \hat \eta &= A^*\big(A_Ix_{0,I}+w-A_I\hat x_I\big) \\
			&=  A^* ( \Id - A_I A_I^+ ) w + \la \eta_F.
	\end{align*}
The residual formula also gives $\hat\eta_I=s_I$. Write $\delta=\norm{A^*w}_\infty$, $L=\norm{A_{I^c}^*A_I}_\infty$, $R=KL+1$, and $S=1-\norm{\eta_{F,I^c}}_\infty>0$. Then
\eql{\label{eq-supp-stab-proof-2}
 \lambda\norm{\hat\eta_{I^c}}_\infty
 \leq R\delta+\lambda\norm{\eta_{F,I^c}}_\infty
 <\lambda\quad\text{if}\quad R\delta<S\lambda.
}
Together with sign consistency and injectivity of $A_I$, this strict inequality proves uniqueness.

    \noindent\textit{Conclusion.} Combining~\eqref{eq-supp-stab-proof-1} and~\eqref{eq-supp-stab-proof-2}, we obtain uniqueness of $\hat x$ whenever the parameters $(\la,w)$ satisfy
	\eq{
		\Rr = \enscond{
				(\de,\la)}{\de + \la < \frac{T}{K}
				\qandq
				R \de - S \la <0
		}
	}
	
	
    The triangular region $\Rr$ contains the simpler triangle
	\eql{\label{eq-critrion-sign-stability}
		\tilde\Rr = \enscond{
				(\de,\la)}{
				\frac{\de}{\la} < \frac{S}{R}
				\qandq
				\la < \la_{\max} 
		}
		\qwhereq
		\la_{\max} \eqdef \frac{T(KL+1)}{K(R+S)}.
	}
	% A_{I^c}^* ( A_I A_I^+ - \Id ) = A_{I^c}^* A_I (A_I^*A_I)^{-1} A_I^* - A_{I^c}^*
\end{proof}

	\begin{figure}[tbp]
\centering
\BookDrawing{chapter-theory-of-sparse-regularization/sparse-theory-recovery-region/original/sparse-theory-recovery-region.pdf}
\caption{Region in the $(\la,\de)$ plane where sign consistency holds.}
\label{fig-review-sparse-theory-recovery-region}
\end{figure}
	
Theorem~\ref{thm-support-stable} requires small correlated noise $\norm{A^*w}_\infty$ and a small $\lambda$, with their ratio controlled, so that regularization does not eliminate the nonzero coefficients of $x_0$. Theorem~\ref{thm-linrate-l1}, by contrast, applies at any noise level $\norm{w}$, but gives neither support control nor uniqueness of $x_\la$, and its constants can be less favorable.

The proof provides explicit constants in terms of three parameters $K,L,S$:
\begin{rs}
	\item $K$ accounts for the conditioning of the operator on the support $I$;
	\item $L$ is the largest sum of absolute correlations between an off-support atom and all atoms on the support;
    \item $S$ measures the margin by which $\eta_F$ avoids saturation outside the support.
\end{rs}
The bounds on $\norm{A^*w}_\infty/\la$ and on $\la$ are given by~\eqref{eq-critrion-sign-stability}. For fixed $\gamma>0$ and sufficiently small $\delta>0$, choosing $\lambda=(1+\gamma)R\delta/S$ gives
\eq{
 \norm{x_0-x_\lambda}_\infty\leq K\left(1+(1+\gamma)\frac{R}{S}\right)\delta.
}
The strict factor $1+\gamma$ ensures the required strict inequality. Such a choice is primarily theoretical because its constants involve the unknown support.

For a class of signals $x_0$, analyzing $\ell^1$ support recovery therefore starts with testing the Fuchs precertificate $\eta_F$: feasibility requires $\norm{\eta_F}_\infty \leq 1$, and support stability requires a strict bound outside the support.
%
Figure~\ref{fig-certif-cs} illustrates the behavior for random $A$: the precertificate $\eta_F$ becomes nondegenerate when $P$ is sufficiently large. Section~\ref{sec-cs-certif} analyzes this phenomenon.

\subsection{Sparsistency for Arbitrary Noise}

The small-noise theorem fixes the sign of the solution in advance. To obtain support containment for arbitrary noise levels, one can instead control all sign patterns on a prescribed support $I$. This gives the exact recovery coefficient of Tropp~\cite{Tropp2006JustRelax}:
\eq{
 \operatorname{ERC}(I)=1-\max_{j\notin I}\norm{A_I^+a_j}_1.
}
The maximum over an empty complement is taken to be zero. Assume $A_I$ is injective and $\operatorname{ERC}(I)>0$. Let $P_I=A_IA_I^+$ and choose $\lambda>0$ so that
\eq{
 \norm{A_{I^c}^*(\Id-P_I)y}_\infty
 <\lambda\operatorname{ERC}(I).
}
Then the Lasso has a unique solution supported within $I$; some coefficients may vanish, so equality of supports is not asserted.

Indeed, minimize the Lasso objective over vectors supported on $I$. The resulting vector $\hat x_I$ is unique, since $A_I$ is injective. Its optimality condition gives $q_I\in\partial\norm{\cdot}_1(\hat x_I)$ and
\eq{
 y-A_I\hat x_I=(\Id-P_I)y+\lambda A_I(A_I^*A_I)^{-1}q_I.
}
For $j\notin I$, the corresponding dual coefficient is bounded by
\eq{
 \frac{|\langle a_j,(\Id-P_I)y\rangle|}{\lambda}
 +\norm{A_I^+a_j}_1<1.
}
Thus the restricted minimizer satisfies the full optimality conditions with strict inequality outside $I$, and injectivity proves uniqueness. If $y=Ax_0+w$ and $\supp(x_0)\subset I$, the orthogonal residual above is $(\Id-P_I)w$. The condition therefore states explicitly how large $\lambda$ must be relative to the noise.

\section{Sparse Deconvolution Case Study}

For random $A$, Chapter~\ref{chap-cs} gives probabilistic conditions under which $\eta_F$ is a valid certificate.

Super-resolution exposes a limitation of the Fuchs construction. The columns $(a_i)_i$ of $A$ are often samples of a smooth kernel, so nearby columns are highly correlated.

We consider $a_i=\phi(z_i)$, where $(z_i)_i \subset \XX$ is a sampling grid in a domain $\XX$ and $\phi : \XX \rightarrow \Hh$ is a smooth map. One has
\eq{
	A x = \sum_i x_i \phi(z_i).
}
The coefficients of a sparse vector $x$ are the weights of the discrete measure $m_{x} \eqdef \sum_{i=1}^N x_i \de_{z_i}$, whose atoms lie on the reconstruction grid.


The matrix $A$ discretizes an operator on Radon measures, $\Aa : m \in \Mm(\XX) \mapsto y = \Aa m \in \Hh$, defined by
\eq{
	\Aa(m) \eqdef \int_\XX \phi(x) \d m(x).
}
For a discrete measure, the two representations agree: $\Aa(m_x)=A x$.

For example, take $\Hh=L^2(\XX)$ on $\XX=\RR^d$ or $\XX=\TT^d$ and let $\phi(z) = \tilde\phi(\cdot-z)$. The resulting operator is convolution:
\eq{
	(\Aa m)(z) = \int \tilde\phi(z-x) \d m(x) = (\tilde \phi \star m)(z).
}
The observation space $\Hh$ is infinite-dimensional in this example. Sampling the output instead gives $\phi(z) = (\tilde\phi(r_j-z))_{j=1}^P$. The observation grid $r \in \XX^P$ need not coincide with the reconstruction grid $z \in \XX^N$.




\begin{figure}[tbp]
\centering
\BookDrawing{chapter-theory-of-sparse-regularization/spikes/original/sparse-theory-convolution-spikes.pdf}
\caption{Convolution operator.}
\label{fig-spikes}
\end{figure}



A closely related example uses $\phi(z)=(e^{\imath k z})_{k=-f_c}^{f_c}$ on $\XX=\TT$, so that $\Aa$ corresponds to computing the $2f_c+1$ low-frequency coefficients of the Fourier transform of the measure
\eq{
	\Aa(m) = \pa{ \int_\TT e^{\imath k x} \d m(x) }_{k=-f_c}^{f_c}.
}
The operator $\Aa^*\Aa$ is a convolution against an ideal low-pass (Dirichlet) kernel. By weighting the Fourier coefficients, one can model low-pass filters on the torus.

On $\XX=\RR^+$, another example is the Laplace transform:
\eq{
	\Aa(m) = z \mapsto \int_{\RR^+} e^{-x z} \d m(x).
}

Define the continuous covariance kernel by
\eq{
	\foralls (z,z') \in \XX^2, \quad
	\Cc(z,z') \eqdef \dotp{\phi(z)}{\phi(z')}_{\Hh}.
}
The function $\Cc$ is the kernel of $\Aa^* \Aa$.
%
The discrete covariance, defined on the computational grid, is $C=(\Cc(z_i,z_{i'}))_{i,i'} \in \RR^{N \times N}$, while its restriction to some support set $I$ is $C_{I,I}=(\Cc(z_i,z_{i'}))_{(i,i')\in I^2}\in\RR^{|I|\times|I|}$.

By~\eqref{eq-eta-f-cov}, the vector $\eta_F$ samples the continuous precertificate $\tilde\eta_F$ on the grid:
\eq{
	\eta_F = ( \tilde\eta_F(z_i) )_{i=1}^N \in \RR^N, 
}
\eql{\label{eq-etaf-cont}
	\qwhereq \tilde\eta_F(x) = \sum_{i \in I} b_i \Cc(x,z_i) \qwhereq b_I = C_{I,I}^{-1} \sign(x_{0,I}), 
}
Thus $\eta_F$ samples a linear combination of $|I|$ kernel functions $(\Cc(x,z_i))_{i \in I}$.

The question is whether $\norm{\eta_F}_{\ell^\infty} \leq 1$. Along increasingly dense grids, continuity implies that a uniform bound on the sampled certificate requires $\norm{\tilde\eta_F}_{L^\infty}\leq1$. A finite grid may miss an overshoot between samples. The construction constrains $\tilde\eta_F$ only to interpolate $\sign(x_{0,i})$ at support points $z_i$; it does not prevent the function from leaving $[-1,1]$ nearby. Figure~\ref{fig-certif-convol} illustrates this fact.


In one dimension, a certificate bounded in magnitude by one must satisfy $\eta'(z_i)=0$ at every interior support point $i \in I$, where it already interpolates the sign. Adding these necessary conditions gives the minimum-norm precertificate with vanishing derivatives:
\eql{\label{eq-vanishing-precertif}
 \tilde\eta_V=\Aa^*p_V,\qquad
 p_V=\uargmin{p\in\Hh}\enscond{\norm{p}_\Hh}{
 (\Aa^*p)(z_i)=\sign(x_{0,i}),\quad
 (\Aa^*p)'(z_i)=0\quad(i\in I)}.
}
Here we restrict to a one-dimensional domain and assume that the interpolation constraints are independent. The adjoint is $(\Aa^*p)(z)=\langle\phi(z),p\rangle_\Hh$, and $\eta_V=(\tilde\eta_V(z_i))_{i=1}^N$. A derivative constraint alone does not ensure $|\tilde\eta_V|\leq1$; this bound must be checked separately.
As in~\eqref{eq-etaf-cont}, this precertificate is a linear combination of kernel functions, now with $2|I|$ terms:
\eq{
	\tilde\eta_V(x) = \sum_{i \in I} b_i \Cc(x,z_i) + c_i \partial_2\Cc(x,z_i), 
}
where $\partial_2\Cc$ is the derivative of $\Cc$ with respect to the second variable, and $(b,c)$ solve a $2|I| \times 2|I|$ linear system
\eq{
	\begin{pmatrix} b \\ c \end{pmatrix} 
	= 
	\begin{pmatrix}
		(\Cc(z_i,z_{i'}))_{i,i' \in I^2} & (\partial_2\Cc(z_i,z_{i'}))_{i,i' \in I^2} \\
		(\partial_1\Cc(z_i,z_{i'}))_{i,i' \in I^2} & (\partial_1\partial_2\Cc(z_i,z_{i'}))_{i,i' \in I^2}
	\end{pmatrix}^{-1}
	\begin{pmatrix} \sign(x_{0,I}) \\ \zeros_I \end{pmatrix} .
}
%
Evaluating $\tilde\eta_V=\Aa^*p_V$ on the reconstruction grid gives the discrete precertificate $\eta_V$.
%
Figure~\ref{fig-certif-convol} shows better behavior of $\eta_V$ than of $\eta_F$. If $\norm{\eta_V}_\infty\leq1$ and $A$ is injective on its saturation set, Theorem~\ref{thm-linrate-l1} gives a linear rate in the grid-based $\ell^2$ norm. As the grid is refined, however, this coefficient $\ell^2$ norm does not provide an intrinsic distance between spike measures, which need not have $L^2$ densities.
%
When $\eta_V$ differs from $\eta_F$, one cannot directly apply Theorem~\ref{thm-support-stable}: exact support stability on increasingly fine grids can fail in super-resolution problems.
%
Methods without a fixed grid provide a more suitable framework for such stability results. In a grid-free formulation, a valid vanishing-derivative precertificate coincides with the minimum-norm certificate under the corresponding interpolation and nondegeneracy assumptions. This leads to stability results for spike locations and amplitudes, rather than a fixed-grid coefficient norm.
