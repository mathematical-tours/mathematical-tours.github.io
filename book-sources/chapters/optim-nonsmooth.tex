\chapter{Nonsmooth Convex Optimization}
\label{chap-conv-duality}

\chapteroverview{%
Many recovery and learning problems combine smooth losses with constraints or penalties that are not differentiable.
Solving them efficiently requires understanding both the cost of an iteration and the conditions for convergence.
We begin with subgradient, projection, and barrier methods, then develop proximal maps and splitting algorithms, including ADMM and primal--dual schemes.}

The main references for this chapter are~\cite{chambolle2010introduction,chambolle2016introduction,boyd2004convex}. Further background is provided in~\cite{parikh2014proximal,boyd2011distributed,beck2014introduction}.

We consider a general convex optimization problem
\eql{\label{eq-nonsmooth-general-pbm}
	\umin{x \in \Hh} f(x)
}
where $\Hh=\RR^p$ is a finite-dimensional Euclidean space,
and seek algorithms with a low cost per iteration. The first-order methods below use gradients, subgradients, or proximal maps. Unless stated otherwise, functions are proper, lower semicontinuous, and convex, and a minimizer is assumed to exist.

\section{Descent Methods}
\label{sec-grad-descent}

We extend the gradient method of Section~\ref{sec-grad-desc-basic} to nonsmooth objectives and constraints.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Gradient Descent}

The optimization problem~\eqref{eq-ip-tv-eps} is an unconstrained problem of the form~\eqref{eq-nonsmooth-general-pbm} with a smooth objective: $f : \Hh \rightarrow \RR$ is continuously differentiable with Lipschitz gradient. Here $\nabla f(x)\in\Hh$ is the gradient of the objective with respect to its variable $x$. It is distinct from the spatial gradient $\nabla x$ of a signal or image, which is a vector field. The functional gradient satisfies the first-order expansion
\eq{
	f(x+r) = f(x) + \dotp{\nabla f(x)}{r}_{\Hh} + O(\norm{r}_{\Hh}^2)
}
The Lipschitz-gradient assumption gives the quadratic remainder $O(\norm{r}_{\Hh}^2)$; differentiability alone gives $o(\norm{r}_{\Hh})$. Examples of gradient computations appear in Section~\ref{eq-example-grad}.

For such a function, the gradient descent algorithm is defined as
\eql{\label{eq-nonsmooth-grad-desc}
	\iit{x} \eqdef \it{x} - \tau_\ell \nabla f( \it{x} ).
}
The step size $\tau_\ell>0$ balances stability against progress per iteration.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Subgradient Method}

For a nonsmooth objective $f$, replace the gradient in~\eqref{eq-nonsmooth-grad-desc} by a subgradient:
\eql{\label{eq-subgrad-desc}
	\iit{x} \eqdef \it{x} - \tau_\ell \it{g}
	\qwhereq
	\it{g} \in \partial f( \it{x} ).
}
A subgradient step need not decrease the objective. Convergence generally requires diminishing step sizes, as the example $f(x)=|x|$ illustrates. This can make the method slow. When the objective admits a useful decomposition, the proximal methods developed below can exploit it more effectively.

\begin{thm}
Assume $f$ is convex, a minimizer $x^\star$ exists, and the chosen subgradients satisfy $\norm{g^\ell}\leq G$. If $\sum_\ell\tau_\ell=+\infty$ and $\sum_\ell\tau_\ell^2<+\infty$, then the iterates converge to a minimizer. More generally,
\eq{\min_{0\leq\ell<N}\{f(x^\ell)-f(x^\star)\}
\leq\frac{\norm{x^0-x^\star}^2+G^2\sum_{\ell<N}\tau_\ell^2}{2\sum_{\ell<N}\tau_\ell}.}
For a prescribed horizon $N$ and positive bounds $D,G$, the constant step $\tau_\ell=D/(G\sqrt N)$ gives a bound $DG/\sqrt N$ when $\norm{x^0-x^\star}\leq D$. A step $1/(\ell+1)$ satisfies the convergence conditions but does not, by itself, give an $O(N^{-1/2})$ bound.
\end{thm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Projected Gradient Descent}
\label{sec-proj-grad}

We consider a constrained optimization problem
\eql{\label{eq-constr}
	\umin{x \in \Cc} f(x) 
} 
where $\Cc \subset \RR^S$ is a nonempty closed convex set and $f : \RR^S \rightarrow \RR$ is convex and continuously differentiable. The convergence theorem below additionally assumes a Lipschitz gradient.

To impose the constraint, follow the gradient step~\eqref{eq-nonsmooth-grad-desc} by projection:
\eql{\label{eq-proj-grad-desc}
	\iit{x} \eqdef \Proj_\Cc \pa{ \it{x} - \tau_\ell \nabla f( \it{x} ) }.
}
Here $\Proj_\Cc$ is the orthogonal projection onto $\Cc$, defined by
\eq{
	\Proj_\Cc(x) = \uargmin{x' \in \Cc} \norm{x-x'}
}
The projection is unique because $\Cc$ is nonempty, closed, and convex.
%
Projected gradient retains analogous convergence guarantees under the following assumptions.

\begin{thm}\label{thm-proj-grad}
Assume $f$ is convex with $L$-Lipschitz gradient, $\Cc$ is nonempty, closed, and convex, and a constrained minimizer exists. For a fixed $0<\tau<2/L$, projected gradient descent converges to a constrained minimizer. If $0<\tau\leq1/L$, then for $\ell\geq1$,
\eq{f(x^\ell)-f(x^\star)\leq\frac{\norm{x^0-x^\star}^2}{2\tau\ell}.}
If $f$ is $\mu$-strongly convex, the iterates converge linearly.
\end{thm}
\begin{proof}
This is the special case $g=\iota_\Cc$ of Theorem~\ref{thm-fb-conv}. In the strongly convex case, the gradient step is a contraction for the admissible fixed step sizes, and composition with the nonexpansive projector preserves that property.
\end{proof}

The main practical difficulty in~\eqref{eq-proj-grad-desc} is evaluating the projection. For several sets arising in $\ell^1$ minimization, it can be computed efficiently.

\section{Interior Point Methods}

We briefly describe interior-point methods, which often require relatively few but expensive iterations. They extend Newton-type methods to problems whose constraints have a suitable barrier representation, such as nonnegativity of vectors or positive definiteness of matrices.


To illustrate the main idea, we consider the following problem
\eql{\label{eq-ip-type}
	\umin{x\in \RR^d, A x \leq y} f(x)
}
for $A \in \RR^{m \times d}$.
%
Related formulations replace the vector $A x$ by a matrix and the componentwise inequality $\leq$ by a positive-semidefinite constraint.

\begin{exmp}[Lasso]
	The Lasso problem (applied to the regression problem $Bw \approx b$ for $B \in \RR^{n \times p}$)
	\eql{\label{eq-lasso-primal}
		\umin{w \in \RR^p} \frac{1}{2}\norm{Bw-b}^2 + \la \norm{w}_1
	} 
	with $\la>0$ can be recast as~\eqref{eq-ip-type} by introducing nonnegative variables
	$x=(x_-,x_+) \in \RR^{2p}$ (so that $d=2p$)  and $w = x_+ - x_-$ with $(x_+,x_-) \geq 0$, so that 
	$f(x) = \frac{1}{2}\norm{B(x_+-x_-)-b}^2 + \la \dotp{x}{\ones}$, $A=-\Id_{2p}$, and $y=0$ (so that $m=2p$).
\end{exmp}

\begin{exmp}[Dual of the Lasso]
	Alternatively, solve the dual of~\eqref{eq-lasso-primal}:
	\eql{
		\umin{\norm{B^\top q}_\infty \leq 1} f(q)=\frac{\la}{2}\norm{q}^2-\dotp{q}{b}
	} 
	so that one can use $A=\begin{pmatrix} B^\top \\-B^\top \end{pmatrix}$ and the constraint right-hand side is $\ones_{2p}$ (so that $d=n$ and $m=2p$).
\end{exmp}

Interior-point methods approximate~\eqref{eq-ip-type} by introducing a logarithmic barrier:
\eql{\label{eq-ip-relaxed}
	\umin{x\in\RR^d,\ Ax<y} f_t(x) \eqdef f(x) - \frac{1}{t} \text{Log}( y-A x  )
}
where 
\eq{
	\text{Log}(u) \eqdef \sum_i \log(u_i)
}
The function $-\mathrm{Log}$ is strictly convex and diverges at the boundary of the positive orthant.
%
Assuming strict feasibility and existence of central-path minimizers $x(t)$, the objective values $f(x(t))$ approach the original optimal value as $t\to+\infty$.

For a fixed $t$, assume $f_t$ is twice continuously differentiable with positive definite Hessian along the iterates. Apply Newton's method to~\eqref{eq-ip-relaxed}, using a line search such as Armijo backtracking~\eqref{eq-armijo-rule} to choose $0<\tau_\ell\leq1$ while maintaining $A\it{x}<y$:
\eql{\label{eq-ip-newton}
	\iit{x} \eqdef \it{x} - \tau_\ell [ \partial^2 f_t(\it{x})]^{-1} \nabla f_t(\it{x})
}
The gradient and Hessian are
\eq{
	\nabla f_t(x) = \nabla f(x) + \frac{1}{t} A^\top \frac{1}{y-Ax}
	\qandq
	\partial^2 f_t(x) = \partial^2 f(x) + \frac{1}{t} A^\top \diag\pa{\frac{1}{(y-Ax)^2}} A.
}
A natural stopping criterion for the inner Newton iterations is
\eq{
	\dotp{[ \partial^2 f_t(\it{x})]^{-1} \nabla f_t(\it{x})}{\nabla f_t(\it{x})} < \frac{\epsilon}{2}.
}
% which ensures that.


The barrier method approximately minimizes $f_t$ through~\eqref{eq-ip-newton}, tracing the central path $t \mapsto x(t)$ for increasing parameters $t=t_k=\mu^k t_0$ with $\mu > 1$.
%
A warm start makes this continuation strategy efficient: initialize the Newton iterations~\eqref{eq-ip-newton} for $x(t_k)$ at the preceding solution $x(t_{k-1})$.
%
Along the central path, the logarithmic barrier gives $f(x(t_k))-f(x^\star) \leq m/t_k$, where $m$ is the number of scalar constraints. To reach error $\epsilon$, take $k=0, \ldots, K$ with
\eq{
	\frac{m}{t_{K}} = \frac{m}{t_0 \mu^{K}} \leq \epsilon.
}
Thus $O(|\log(\epsilon)|)$ barrier-parameter updates suffice for accuracy $\epsilon$. The total cost also depends on the number of inner Newton iterations~\eqref{eq-ip-newton}, whose control requires further assumptions on $f$.


The relevant complexity theory uses self-concordant barriers. A convex function $\phi$ on a line is self-concordant when
\eq{|\phi^{(3)}(s)|\leq2\phi''(s)^{3/2}.}
The logarithmic barrier has this property. For a linear objective, the appropriately scaled objective $t f-\mathrm{Log}(y-Ax)$ is self-concordant; arbitrary positive rescaling does not preserve the same self-concordance constant. With a suitable path-following neighborhood and barrier-parameter schedule, the number of Newton steps can be bounded polynomially in the problem parameters and logarithmically in $1/\epsilon$. The bound depends on the barrier parameter (equal to $m$ for this logarithmic barrier), not only on the requested accuracy. A fixed number of Newton steps per outer iteration cannot be asserted for an arbitrary update factor $\mu>1$.

\section{Proximal Algorithm}

For a nonsmooth objective $f$, an explicit gradient step may be undefined. An implicit step remains available even when $f$ is nonsmooth.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Proximal Map}

For a step size $\tau>0$, define the proximal map by
\eql{\label{eq-defn-proximal}
	\foralls x, \quad
	\Prox_{\tau f}(x) \eqdef \uargmin{x'} \frac{1}{2}\norm{x-x'}^2+\tau f(x').
}
It balances a decrease in $f$ against a quadratic penalty for moving away from $x$.
%
Since $f\in\Ga_0(\Hh)$, the objective $\frac12\norm{x-\cdot}^2+\tau f$ is strongly convex and coercive, so $\Prox_{\tau f}$ is well defined and single-valued.

For an indicator $f=\iota_\Cc$, the proximal map is the projection $\Prox_{\iota_\Cc}=\Proj_{\Cc}$. It therefore generalizes projection to convex functions. For each fixed input, the proximal point is also the projection onto a suitable sublevel set of $f$. Like an orthogonal projection, the proximal map is nonexpansive.

\begin{prop}
	One has $\norm{ \prox_{f}(x)-\prox_{f}(y) } \leq \norm{x-y}$.
\end{prop}

\paragraph{Examples}

Several common proximal maps have closed forms.

\begin{prop}
One has
\eql{\label{eq-prox-l2-l1}
	\Prox_{\frac{\tau}{2} \norm{\cdot}^2}(x) =  \frac{x}{1+\tau}, 
	\qandq
	\Prox_{\tau\norm{\cdot}_1}(x)=\Ss_\tau^1(x),
}
where the soft-thresholding is defined as
\eq{
	\Ss_\tau^1(x) \eqdef ( S_\tau(x_i) )_{i=1}^p \qwhereq
	S_\tau(r)\eqdef\sign(r)(|r|-\tau)_+,
}
(see also~\eqref{eq-soft-thresh}). For $A \in \RR^{P \times N}$, one has
\eql{\label{eq-prox-quad}
	\Prox_{\frac{\tau}{2}\norm{A \cdot-y}^2}(x) = (\Id_N + \tau A^* A)^{-1} ( x + \tau A^* y ).
	%= (\Id_{P} +\tau A^*A)^{-1} A^*	=  A^* (\Id_N + \tau AA^*)^{-1} 
}
\end{prop}
\begin{proof}
	The proximal map of $\norm{\cdot}_1$ was derived informally in Proposition~\ref{prop-equiv-sparse-thresh}.
	%
	For a rigorous proof, use separability to reduce to a scalar input $r$. The optimality condition for its proximal point $z$ is $0 \in z-r + \tau \partial |\cdot|(z)$.
	The candidate $z=0$ is optimal precisely when $0 \in -r + [-\tau,\tau]$, or equivalently $r \in [-\tau,\tau]$.
	%
	For $z \neq 0$, optimality requires $z = r - \tau \sign(z)$. When $r \notin [-\tau,\tau]$,
	the candidate $z = r - \tau \sign(r)$ has the same sign as $r$, so it is the unique minimizer.
	%
	For the quadratic case
	\eq{
		z = \Prox_{\frac{\tau}{2}\norm{A \cdot-y}^2}(x) 
		\quad\Leftrightarrow\quad 
		z-x + \tau A^*( A z - y ) = 0
		\quad\Leftrightarrow\quad
		(\Id_N + \tau A^* A) z = x + \tau A^* y.
	}
\end{proof}


For some nonconvex functions, the proximal minimization also has solutions, but they may be nonunique. For example,
$\Prox_{\tau \norm{\cdot}_0}$ is hard thresholding at $\sqrt{2\tau}$; see Proposition~\ref{prop-equiv-sparse-thresh}. At $|x_i|=\sqrt{2\tau}$, both $0$ and $x_i$ are minimizers.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Basic Properties}

The following identities simplify proximal computations.

\begin{prop}
One has
\eq{
	\Prox_{f+\dotp{y}{\cdot}}=\Prox_f(\cdot-y), \quad
	\Prox_{f(\cdot-y)} = y + \Prox_{f}(\cdot-y).
}
If $f(x)=\sum_{k=1}^K f_k(x_k)$ for $x=(x_1,\ldots,x_K)$ is separable, then
\eql{\label{eq-prox-separable}
	\Prox_{\tau f}(x) = ( \Prox_{\tau f_k}(x_k) )_{k=1}^K.
}
\end{prop}
\begin{proof}
The optimality condition for $z=\Prox_{f+\dotp{y}{\cdot}}(x)$ is $0\in z-x+\partial f(z)+y$, or $x-y\in z+\partial f(z)$. Thus $z=\Prox_f(x-y)$. For the translated function, substitute $w=z-y$ in
$\min_z\{\frac12\norm{z-x}^2+f(z-y)\}$ to obtain $z=y+\Prox_f(x-y)$. Separability follows by minimizing each coordinate block independently.
\end{proof}

Composition with a linear map having orthonormal rows admits an explicit formula.

\begin{prop}\label{prop-prox-tightframe}
	If $A \in \RR^{P \times N}$ satisfies $AA^*=\Id_P$, then
	\eq{
		\Prox_{f \circ A} = A^* \circ \Prox_{f} \circ A + \Id_N - A^* A.
	}
	In particular, if $A$ is orthogonal, then $\Prox_{f \circ A} = A^* \circ \Prox_{f} \circ A$.
\end{prop}

\subsection{Related Concepts}

%%%
\paragraph{Link with the subdifferential.}

For a set-valued map $U : \Hh \hookrightarrow \Gg$, we define the inverse set-valued map $U^{-1} : \Gg \hookrightarrow \Hh$ by 
\eql{\label{eq-inv-setvalued}
	h \in U^{-1}(g) 
	\quad\Longleftrightarrow\quad
	g \in U(h) 
} 

The proximal map is the resolvent of the subdifferential.


\begin{prop}
	One has $\Prox_{\tau f} = (\Id+\tau\partial f)^{-1}$.
\end{prop}
\begin{proof}
The optimality condition gives the equivalences
\eq{
	z = \Prox_{\tau f}(x) 
	\Leftrightarrow
	0 \in z-x+\tau\partial f(z)
	\Leftrightarrow
	x \in (\Id+\tau\partial f)(z)
	\Leftrightarrow
	z = (\Id+\tau\partial f)^{-1}(x)
}	
The final relation is an equality because the inverse is single-valued here.
\end{proof}

This resolvent interpretation connects proximal algorithms with the theory of maximally monotone operators.

%%%%
\paragraph{Link with duality.}

Moreau decomposition relates the proximal maps of a function and its conjugate.

\begin{thm}[Moreau decomposition]
	One has
	\eq{
		\Prox_{\tau f} = \Id - \tau \Prox_{f^* / \tau}( \cdot/\tau ).
	}
\end{thm}

Thus a tractable proximal map for $f$ gives one for $f^*$, and conversely.
%
For example, Proposition~\ref{prop-dual-lp} gives the following expression for soft thresholding
\eq{
	\Prox_{\tau \norm{\cdot}_1}(x) =  x - \tau \Proj_{\norm{\cdot}_\infty \leq 1}( x/\tau )
	 =  x -  \Proj_{\norm{\cdot}_\infty \leq \tau}( x )
	 \qwhereq
	 \Proj_{\norm{\cdot}_\infty \leq \tau}( x ) = \min(\max(x,-\tau),\tau).
}
The minimum and maximum in this clipping formula act componentwise.

For an indicator $f=\iota_{\Cc}$ of a closed convex cone $\Cc$,
\eql{\label{eq-polar-cone}
	(\iota_{\Cc})^* = \iota_{\Cc^\circ} 
	\qwhereq
	\Cc^\circ \eqdef \enscond{y}{ \foralls x \in \Cc, \dotp{x}{y} \leq 0 } 
}	
where $\Cc^\circ$ is the polar cone. This conjugacy relation~\eqref{eq-polar-cone} is specific to cones: for a general convex set $\Cc$, the conjugate of its indicator need not be an indicator.
%
Equation~\eqref{eq-polar-cone} yields Moreau polar decomposition:
\eq{
	x = \Proj_{\Cc}(x) +^\bot \Proj_{\Cc^\circ}(x)
} 
The notation $+^\bot$ indicates that the two summands are orthogonal.
%
For a linear subspace $\Cc=V$, this reduces to the orthogonal decomposition $\RR^p = V \oplus^\bot V^\bot$.

%%%%
\paragraph{Link with Moreau--Yosida regularization.}

A proximal step is also a gradient step on the Moreau--Yosida envelope $f_\mu$ of $f$ from~\eqref{eq-moreau-yosida}.

\begin{prop}
	One has
	\eq{
		\Prox_{\mu f} = \Id - \mu \nabla f_\mu.
	}
\end{prop}

\section{Proximal Gradient Algorithms}

Splitting algorithms exploit a decomposition of the objective into terms whose proximal maps can be computed separately.
%
Different decompositions of the same objective produce different splitting algorithms. A useful decomposition balances the cost of the proximal maps against the convergence of the resulting iteration.
%
Step sizes and other parameters also need to be chosen; theoretical convergence bounds and line-search strategies can guide their selection.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Proximal Point Algorithm}

One has the following equivalence 
\begin{align}\label{eq-proxpoint-fix}
	x^\star \in \argmin f
	&\quad\Leftrightarrow\quad
	0 \in \partial f(x^\star)
	\quad\Leftrightarrow\quad
	x^\star \in (\Id+\tau \partial f)(x^\star) \\
	&\quad\Leftrightarrow\quad
	x^\star = (\Id+\tau \partial f)^{-1}(x^\star) = \Prox_{\tau f}(x^\star).
\end{align}
This shows that being a minimizer of $f$ is equivalent to being a fixed point of $\Prox_{\tau f}$.
%
This fixed-point characterization suggests the proximal point iterations
\eql{\label{eq-proximal-point}
	\iit{x} \eqdef \Prox_{\tau_\ell f}(\it{x}).
}
The proximal point method does not require an upper stability bound on the step size. Steps must nevertheless be large enough in aggregate; the following sufficient condition is convenient.

\begin{thm}
	If $0<\tau_{\min} \leq \tau_\ell \leq \tau_{\max} < +\infty$, then $\it{x} \rightarrow x^\star$ for a minimizer of $f$.
\end{thm}

This implicit step~\eqref{eq-proximal-point} should be compared with a gradient descent step~\eqref{eq-nonsmooth-grad-desc}
\eq{
	\iit{x} \eqdef (\Id-\tau_\ell\nabla f)(\it{x}).
}
The implicit resolvent $(\Id+\tau_\ell\partial f)^{-1}$ replaces the explicit map $\Id-\tau_\ell\nabla f$. For small $\tau_\ell$ and smooth $f$, the two agree to first order. The implicit step also remains well defined for nonsmooth objectives and converges under the assumptions above, without the upper step-size restriction of explicit gradient descent. This resembles the greater stability of implicit Euler integration. The computational tradeoff is that a general proximal map may be as difficult to evaluate as the original minimization problem.


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Forward--Backward Splitting}
\label{sec-fb}

For a general objective, evaluating $\Prox_{\ga f}$ may be as difficult as solving the original problem.
%
A tractable alternative exploits a decomposition into a smooth term and a term with an accessible proximal map:
\eql{\label{eq-fb-split}
	\umin{x} \Ee(x) \eqdef f(x) + g(x)
}
where $g \in \Ga_0(\Hh)$ can be nonsmooth, whereas $f$ is convex and has a Lipschitz gradient.

For this structure, the fixed-point argument~\eqref{eq-proxpoint-fix} becomes
\begin{align*}
 	x^\star \in \argmin f + g
	& \quad\Leftrightarrow\quad
	0 \in \nabla f(x^\star) + \partial g(x^\star) 
	\quad\Leftrightarrow\quad
	x^\star - \tau \nabla f(x^\star) \in (\Id+\tau \partial g)(x^\star) \\
	& \quad\Leftrightarrow\quad
	x^\star = ( \Id + \tau \partial g )^{-1} \circ ( \Id - \tau \nabla f ) (x^\star).
\end{align*}
This fixed point suggests the following algorithm, known as forward--backward splitting
\eql{\label{eq-fb}
		\iit{x} \eqdef \Prox_{\tau_\ell g}\pa{ \it{x} - \tau_\ell \nabla f(\it{x}) }.
}


%%%
\paragraph{Derivation using surrogate functionals.}

A quadratic surrogate gives another derivation of the algorithm and helps explain its convergence.

%
% The difficulty is the presence of the operator $A$ in the $\ldeux$ norm, which makes this problem significantly more difficult than the simple denoising by regularization\eqref{eq-regul-denoising}.

Replace the objective $\Ee(x)$ at the current iterate by a surrogate $\Ee(x,\it{x})$ that is easier to minimize, and define
\eql{\label{eq-surrogate-iter}
	\iit{x} \eqdef \uargmin{x} \Ee(x,\it{x}).
}
Require the surrogate to majorize the objective and agree with it at the current point:
\eql{\label{eq-surrogate-pty}
	\Ee(x) \leq \Ee(x,x')
	\qandq
	\Ee(x,x) = \Ee(x)
}
and $\Ee(x)-\Ee(x,x')$ should be a smooth function.
%
Property~\eqref{eq-surrogate-pty} ensures that $\Ee$ does not increase along the iterations:
\eq{
	\Ee(\iit{x}) \leq \Ee(\it{x})
}
For proximal-gradient surrogates with a fixed $0<\tau<1/L$, sufficient decrease and the optimality conditions imply that every accumulation point is a minimizer. The majorization and touching conditions alone do not establish this conclusion for arbitrary surrogate schemes.

To construct a surrogate for~\eqref{eq-fb-split}, use the quadratic upper bound~\eqref{eq-above-below-quad} supplied by the $L$-Lipschitz gradient of $f$:
\eq{
	f(x) 
	\leq 
	f(x') + \dotp{\nabla f(x')}{x-x'} + \frac{L}{2}\norm{x-x'}^2,
}
so that for $0 < \tau < \frac{1}{L}$, the function 
\eql{\label{eq-surrogate-ip}
	\Ee(x,x') \eqdef f(x') + \dotp{\nabla f(x')}{x-x'} + \frac{1}{2\tau}\norm{x-x'}^2 + g(x)
}
satisfies the surrogate conditions~\eqref{eq-surrogate-pty}.
%
Minimizing this surrogate is exactly a proximal-gradient step.

\begin{prop}
	The update~\eqref{eq-surrogate-iter} for the surrogate~\eqref{eq-surrogate-ip} is exactly~\eqref{eq-fb}.
\end{prop}
\begin{proof}
	This follows from the fact that
	\eq{
		\dotp{\nabla f(x')}{x-x'} + \frac{1}{2\tau}\norm{x-x'}^2
		= \frac{1}{2\tau} \norm{ x - (x'-\tau \nabla f(x'))  }^2 + \text{cst}.
	}
\end{proof}

%%%
\paragraph{Convergence of FB. }

The majorization argument uses $\tau\leq1/L$, whereas convergence of the iterates holds on the larger interval $0<\tau<2/L$.

\begin{thm}\label{thm-fb-conv}
Let $f$ be convex with $L$-Lipschitz gradient and $g\in\Ga_0(\Hh)$, and assume $\argmin(f+g)\neq\emptyset$. For a fixed step $0<\tau<2/L$, the forward--backward iterates converge to a minimizer. If $0<\tau\leq1/L$, then
\eq{(f+g)(x^\ell)-\min(f+g)\leq\frac{\norm{x^0-x^\star}^2}{2\tau\ell},\qquad\ell\geq1.}
If $f$ is $\mu$-strongly convex, the iterates converge linearly. In particular, for $0<\tau\leq1/L$,
$\norm{x^{\ell+1}-x^\star}\leq(1-\tau\mu)\norm{x^\ell-x^\star}$.
\end{thm}

Taking $g=\iota_{\Cc}$ in~\eqref{eq-fb} recovers projected gradient descent~\eqref{eq-proj-grad-desc}, because the proximal map of an indicator is the projection onto the constraint set.

Applying~\eqref{eq-fb} efficiently requires a tractable proximal map $\Prox_{\tau g}$. A closed form is available for several useful penalties, including the $\ell^1$ penalty used by ISTA in Section~\ref{sec-ista}.

\section{Primal-Dual Algorithms}

Convex duality, developed in Section~\ref{sec-cvx-duality}, suggests new algorithms and allows existing methods to be applied to dual formulations.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Forward--Backward Splitting on the Dual}
\label{sec-fb-dual}

As a first example, apply a familiar method to the Fenchel--Rockafellar problem~\eqref{eq-fench-rock-basicpb}:
\eql{\label{eq-splitting-primal-dual}
	p^\star = \uinf{x} f(x) + g(A x), 
}
Assume additionally that $f$ is $\mu$-strongly convex, and, for simplicity, that $f$ and $g$ are finite and continuous everywhere. Even if $f$ is smooth, applying forward--backward splitting directly to the primal problem requires $\Prox_{\tau(g\circ A)}$. This map can be difficult to evaluate despite having a simple $\Prox_{\tau g}$. Proposition~\ref{prop-prox-tightframe} gives an explicit formula in the special case $AA^*=\Id$.

\begin{exmp}[TV denoising]
Chambolle~\cite{chambolle-algo-tv} developed this approach for total variation denoising:
\eql{\label{eq-tv-denoising}
	\umin{x} \frac{1}{2}\norm{y-x}^2 + \la \norm{\nabla x}_{1,2}
}
where $\nabla x \in \RR^{N \times d}$ is the discrete gradient of a signal ($d=1$) or image ($d=2$) with $N$ spatial samples. The mixed $\ell^1$--$\ell^2$ norm sums the Euclidean norms of the vectors $v_i\in\RR^d$:
\eq{
	\norm{v}_{1,2} \eqdef \sum_i \norm{v_i}.
}	
Here 
\eq{
	f=\frac{1}{2}\norm{\cdot-y}^2
	\qandq
	g=\la\norm{\cdot}_{1,2}
}
Thus $f$ is strongly convex with constant $\mu=1$, and $A=\nabla$ is the discrete gradient operator.
\end{exmp}

\begin{exmp}[Support Vector Machine]
	We consider a supervised classification problem with data $(a_i \in \RR^p, y_i \in \{-1,+1\})_{i=1}^n$.
	%
	With regularization parameter $\lambda>0$, the support vector machine objective is
	\eql{\label{eq-svm}
		\umin{x\in\RR^p}\sum_i\ell(-y_i\dotp{x}{a_i}) + \frac{\lambda}{2}\norm{x}^2
		= \frac{\lambda}{2}\norm{x}^2 + g(A x)
	} 
	where $\ell(s) = \max(0,s+1)$ is the hinge loss and
	the notation is $A_{i,j}=-y_i a_i[j]$ and $g(u)=\sum_i\ell(u_i)$.
	%
	It corresponds to the split~\eqref{eq-splitting-primal-dual} using $f(x) = \frac{\lambda}{2}\norm{x}^2$. 
\end{exmp}


Continuity supplies the qualification for Fenchel--Rockafellar duality, Theorem~\ref{thm-fenchel-Rockafellar}, giving
\eql{\label{eq-dual-split-fAg}
	p^\star = \usup{u}  - g^*(u) - f^*(-A^*u).
}
Strong convexity of $f$ with constant $\mu$ makes the conjugate $f^*$ differentiable with $1/\mu$-Lipschitz gradient. Apply forward--backward splitting~\eqref{eq-fb} to the negative dual objective:
\eq{
	\iit{u} = \Prox_{\tau g^*}\pa{ \it{u} + \tau A \nabla f^*(-A^*\it{u}) }.
}
Choose a fixed step $0<\tau<2/L$, where $L>0$ bounds the Lipschitz constant of the smooth dual gradient; $L=\norm{A}^2/\mu$ is valid when $A\neq0$. The fixed-step assumption ensures that Theorem~\ref{thm-fb-conv} applies.

Once a dual maximizer $u^\star$ is computed, equivalently a minimizer of the negative dual objective, the primal-dual relations~\eqref{eq-primal-dual} recover the unique primal minimizer:
\eq{
	-A^* u^\star \in \partial f(x^\star)  
	\quad\Leftrightarrow\quad
	x^\star \in (\partial f)^{-1}( -A^* u^\star ) = \partial f^*( -A^* u^\star )
	\quad\Leftrightarrow\quad
	 x^\star = \nabla f^*( -A^* u^\star )
}
using differentiability of $f^*$.


\begin{exmp}[TV denoising]
In the particular case of the TV denoising problem~\eqref{eq-tv-denoising}, one has
\eq{
	g^* = \iota_{\norm{\cdot}_{\infty,2} \leq \la}
	\qwhereq
	\norm{v}_{\infty,2} \eqdef \umax{i}\norm{v_i}
}
\eq{
	f^\star(h) = \frac{1}{2}\norm{h}^2+\dotp{h}{y}, 
}
The dual, expressed as a minimization, is therefore
\eq{
	\umin{ \norm{v}_{\infty,2}\leq \la}\frac{1}{2}\norm{ -\nabla^* v + y }^2. 
}
Projected gradient descent, a special case of forward--backward splitting, gives
\eq{
		\Prox_{\tau g^*}(u)=\left(\frac{u_i}{\max\{1,\norm{u_i}/\la\}}\right)_i
	\qandq
	\nabla f^\star(h) = h + y.
}	
Furthermore, one has $\mu=1$ and $A^*A=-\Delta$ for the usual negative-semidefinite discrete Laplacian. With unit grid spacing, standard forward differences satisfy $\norm{A}^2\leq4d$; equality depends on the boundary conditions and grid size.
\end{exmp}


\begin{exmp}[Support Vector Machine]
	For the SVM problem~\eqref{eq-svm}, one has 
	\eq{
		\foralls u \in \RR^n, \quad g^\star(u) = \sum_i \ell^\star(u_i)
	}
	where 
	\eq{
		 \ell^\star(t) = \usup{s} t s - \max(0,s+1)
		 = -t + \usup{s}ts-\max(0,s)
		 = -t + \iota_{[0,1]}(t)
	}
	because the conjugate of $\max(0,s)$ is the indicator of its subdifferential at zero, namely $[0,1]$.
	%
	Also, one has $f(x) = \frac{\lambda}{2}\norm{x}^2$
	so that 
	\eq{
		f^\star(z) = \lambda (\norm{\cdot}^2/2)(z/\lambda) = \lambda \frac{\norm{z/\lambda}^2}{2} = 
		\frac{\norm{z}^2}{2\la}.
	}
	%
	The dual problem reads
	\eq{
		\umin{ 0 \leq v \leq 1 } - \dotp{\ones}{v} + \frac{\norm{A^\top v}^2}{2\la}
	}
	which can be solved by forward--backward splitting, here equivalent to projected gradient descent.
	%
	One thus has 
	\eq{
		\Prox_{\tau g^*}(u)=\uargmin{v\in[0,1]^n}\left\{\frac12\norm{u-v}^2-\tau\dotp{\ones}{v}\right\}
=\Proj_{[0,1]^n}(u+\tau\ones)
=\min\{\max\{u+\tau\ones,0\},\ones\}.
 }
\end{exmp}

\subsection{Douglas--Rachford Splitting}
\label{sec-dr}

We consider here the structured minimization problem
\eql{\label{eq-struct-dr}
	\umin{x \in \RR^p} f(x) + g(x), 
}
Unlike the setting of Section~\ref{sec-fb}, $f$ need not be smooth. We assume that the proximal maps of both $f$ and $g$ are tractable.

\begin{exmp}[Constrained Lasso]
Douglas--Rachford splitting applies, for example, to the noiseless constrained Lasso problem~\eqref{eq-lasso-constr-ip}:
\eq{
	\umin{Ax=y} \norm{x}_1
}
Take $f=\iota_{\Cc_y}$ with $\Cc_y \eqdef \enscond{x}{Ax=y}$, and $g=\norm{\cdot}_1$.
%
As noted in Section~\ref{eq-linearprog-lasso}, this problem is equivalent to a linear program.
%
The proximal map of $g$ is soft thresholding~\eqref{eq-prox-l2-l1}. The proximal map of $f$ is projection onto the affine space $\Cc_y$, obtained by solving the linear system in~\eqref{eq-proj-aff}. This projection is particularly efficient for inpainting and for deconvolution operators diagonalized by the Fourier transform.
\end{exmp}

The Douglas--Rachford iterations read
\eql{\label{eq-dr-iter} 
	 \iit{\tilde x} \eqdef \pa{1-\frac{\mu}{2}} \it{\tilde x} + 
	\frac{\mu}{2} \rProx_{\tau g}( \rProx_{\tau f}( \it{\tilde x} )  ) 
	\qandq \iit{x} \eqdef \Prox_{\tau f}( \iit{\tilde x}), 
}
where the reflected proximal map is
\eq{   	
	\rProx_{\tau f}(x) = 2\Prox_{\tau f}(x)-x .
}
Assume $0\in\partial f(x^\star)+\partial g(x^\star)$ for some $x^\star$ (for example, a minimizer exists and the relative interiors of the domains intersect). Then, for any $\tau>0$, any $0 < \mu < 2$, and any $\tilde x_0$, the shadow iterates $\it{x}$ converge to a minimizer of $f+g$, which may depend on the initialization.

Interchanging $f$ and $g$ gives another valid iteration.

%%%
\paragraph{More than two functions.}

A product-space formulation treats $K$ functions $(f_k)_k$ symmetrically:
\eq{
	\umin{x} \sum_k f_k(x) = \umin{X=(x_1,\ldots,x_K)} f(X)+g(X)
	\qwhereq
	f(X) = \sum_k f_k(x_k)
	\qandq
	g(X)=\iota_{\Delta}(X)
}
where $\Delta=\enscond{X}{x_1=\ldots=x_K}$ is the diagonal. The proximal operator of $g$ is
\eq{
	\Prox_{\tau g}(X)=\Proj_\Delta(X)=(\bar x,\ldots,\bar x)
	\qwhereq
	\bar x = \frac{1}{K}\sum_k x_k
}
The proximal map of $f$ separates into the maps of $(f_k)_k$ by~\eqref{eq-prox-separable}. Douglas--Rachford then applies through~\eqref{eq-dr-iter}.

%%%
\paragraph{Handling a linear operator.}

For an objective of the form~\eqref{eq-splitting-primal-dual}, introduce an auxiliary variable:
\eq{
	\uinf{x} f_1(x) + f_2(Ax) = \uinf{z=(x,y)} f(z)+g(z)
	\qwhereq
	\choice{
		f(z)=f_1(x)+f_2(y) \\
		g(z)=\iota_\Cc(x,y),
	}
}
where $\Cc=\enscond{(x,y)}{Ax=y}$. Douglas--Rachford splitting applies because the proximal map of $f$ separates into those of $f_1$ and $f_2$ by~\eqref{eq-prox-separable}. The proximal map of $g$ is projection onto $\Cc$. The following proposition gives two equivalent formulas, allowing a choice between systems in the input and output spaces of $A$.

\begin{prop}One has
\eql{\label{eq-proj-axy}
	\Proj_{\Cc}(x,y) = (x-A^*\tilde y,y+\tilde y) = (\tilde x,A\tilde x)
	\qwhereq
	\choice{
		\tilde y \eqdef (\Id_P+AA^*)^{-1}(Ax-y) \\
		\tilde x \eqdef (\Id_N+A^*A)^{-1}(A^*y+x).
	}
}
\end{prop}
\begin{proof}
Minimizing $\frac12\norm{z-x}^2+\frac12\norm{Az-y}^2$ gives $(\Id_N+A^*A)z=x+A^*y$, hence $z=\tilde x$. Alternatively, the Lagrange multiplier for $Az=w$ satisfies $(\Id_P+AA^*)\tilde y=Ax-y$, with $z=x-A^*\tilde y$ and $w=y+\tilde y$.
\end{proof}

\begin{rem}[Inverting structured linear operators]\label{rem-inv-lin}
	Projection computations often require the inverse of a matrix of the form
	$AA^*$, $A^*A$, $\Id_P+AA^*$ or $\Id_N+A^*A$ (see for instance~\eqref{eq-proj-axy}).
	The structure of the operator can make this inexpensive.
	For inpainting, $AA^*$ is diagonal. For deconvolution or partial Fourier measurements such as MRI, $A^*A$ is diagonalized by the Fourier transform.
	%
	When inversion is expensive, primal-dual methods can replace it by applications of $A$ and $A^*$, trading cheaper iterations for a potentially larger iteration count.
\end{rem}

\subsection{Alternating Direction Method of Multipliers}

Douglas--Rachford splitting, introduced in Section~\ref{sec-dr}, can minimize $f+g\circ A$ without strong convexity of $f$. Its direct application requires the proximal maps $\Prox_{\tau f}$ and $\Prox_{\tau g\circ A}$.
%
The alternating direction method of multipliers (ADMM) is related to Douglas--Rachford splitting on the dual problem~\eqref{eq-dual-split-fAg}. This formulation uses $\Prox_{\tau g^*}$, computable from $\Prox_{\tau g}$ by Moreau decomposition, and $\Prox_{\tau(f^*\circ A^*)}$. The primal updates derived below express these operations as alternating minimizations.

For $A \in \RR^{n \times p}$, introduce $y=Ax$ as in~\eqref{eq-primal-dual-fench} and scale the multiplier by $\ga>0$. Define
\eq{\Ll((x,y),z)\eqdef f(x)+g(y)+\ga\dotp{z}{Ax-y}.}
The primal problem becomes
\eql{\label{eq-admm-primal-dual}
	\uinf{x \in \RR^p} f(x) + g(Ax) = \uinf{y = Ax} f(x)+g(y) = \uinf{x \in \RR^p,y \in \RR^n} \usup{z \in \RR^n}
		\Ll((x,y),z).
}
Add a quadratic penalty for violating $Ax=y$ to obtain the augmented Lagrangian
\eq{
	\Ll_\ga((x,y),z) \eqdef \Ll((x,y),z) + \frac{\ga}{2}\norm{Ax-y}^2
	= f(x) + g(y) + \ga \dotp{z}{Ax-y} + \frac{\ga}{2}\norm{Ax-y}^2.
}
Replacing $\Ll$ by $\Ll_\ga$ in~\eqref{eq-admm-primal-dual} preserves the constrained problem: the supremum over $z$ enforces $Ax=y$, where the added penalty vanishes.

The ADMM method then updates
\begin{align}
	\label{eq-admm-1}\iit{y} &\eqdef \uargmin{y} \Ll_\ga( (\it{x},y),\it{z}  ), \\
	\label{eq-admm-2}\iit{x} &\eqdef \uargmin{x} \Ll_\ga( (x,\iit{y}),\it{z}  ), \\
	\label{eq-admm-3}\iit{z} &\eqdef \it{z} + A \iit{x}-\iit{y}. 
\end{align}
For comparison, the method of multipliers jointly minimizes the augmented Lagrangian in $x$ and $y$, then updates the multiplier:
\begin{align*}
	(\iit{x},\iit{y}) &\eqdef \uargmin{x,y} \Ll_\ga( (x,y),\it{z}  ), \\
	\iit{z} &\eqdef \it{z} + A \iit{x}-\iit{y}, 
\end{align*}
ADMM instead updates $x$ and $y$ alternately, which often makes the individual subproblems tractable.

Step~\eqref{eq-admm-3} is gradient ascent in the multiplier $z$ with step $1/\ga$. Assume for this calculation that $f$ is differentiable. Combining the $x$ and $z$ updates enforces $\nabla f(\iit{x})+\ga A^\top\iit{z}=0$, the stationarity relation in $x$. Indeed, the optimality condition for~\eqref{eq-admm-2} reads
\eq{
	0 = \nabla_x \Ll_\ga( (\iit{x},\iit{y}),\it{z}  ) = 
	\nabla f(\iit{x}) + \ga A^\top\it{z}+\ga A^\top(A\iit{x}-\iit{y})
	= \nabla f(\iit{x}) + \ga A^\top \iit{z}. 
}

Step~\eqref{eq-admm-1} is a proximal step, since
\begin{align*}
	\iit{y} &= \uargmin{y} 
	g(y) + \ga \dotp{\it{z}}{A\it{x}-y} + \frac{\ga}{2}\norm{A\it{x}-y}^2
	\\
	&=\uargmin{y} \frac{1}{2}\norm{y - A\it{x} - \it{z}}^2 + \frac{1}{\ga} g(y)
	= \Prox_{g/\ga}( A\it{x} + \it{z} ).
\end{align*}
Step~\eqref{eq-admm-2} uses a quadratic penalty composed with $A$:
\eq{\iit{x}\in\uargmin{x}\left\{\frac12\norm{Ax-\iit{y}+\it{z}}^2+\frac1\ga f(x)\right\}
\eqdef\Prox^A_{f/\ga}(\iit{y}-\it{z}).}
Here $\Prox^A_{f/\ga}(u)$ denotes the minimizer set of $\frac12\norm{Ax-u}^2+f(x)/\ga$; it can be set-valued when $A$ is not injective. Under the usual qualification for this subproblem, if
\eq{p=\Prox_{\ga(f^*\circ A^*)}(\ga u),}
then every minimizing $x$ satisfies $Ax=u-p/\ga$ and $A^*p\in\partial f(x)$. If $A$ has full column rank, this gives
\eq{\Prox^A_{f/\ga}(u)=A^+\left(u-\frac1\ga\Prox_{\ga(f^*\circ A^*)}(\ga u)\right).}
For noninjective $A$, applying $A^+$ alone need not recover a minimizer; the component in $\ker(A)$ must also minimize $f$.

\begin{exmp}[Constrained TV recovery]
	We consider the problem of minimizing a TV norm under affine constraints
	\eq{
		\umin{Bx=y} \norm{\nabla x}_1
	}
	which is of the form $f + g \circ A$ when defining $A=\nabla$, $g=\norm{\cdot}_1$ and $f=\iota_{B \cdot = y}$. 
	%
	The proximal map of $g$ is block soft thresholding for isotropic TV (componentwise soft thresholding for anisotropic TV), while the ``twisted'' proximal operator is
	\eq{
		\Prox^A_{f/\ga}(u) = \uargmin{Bx=y} \frac{1}{2}\norm{Ax-u}^2
		=\uargmin{x}\usup{w}\left\{\frac12\norm{\nabla x-u}^2+\dotp{w}{Bx-y}\right\}
	}
	Introducing the multiplier $w$ for $Bx=y$ gives first-order conditions for $(x^\star,w^\star)$ in the form of a linear system:
	\eq{
		\begin{pmatrix}
			\nabla^\top \nabla & B^\top \\
			B & 0
		\end{pmatrix}
		\begin{pmatrix}
			x^\star \\ w^\star
		\end{pmatrix}
		= 
		\begin{pmatrix}
			\nabla^\top  u \\ y 
		\end{pmatrix}
	}
	Here $\Delta=-\nabla^\top \nabla$ is the discrete Laplacian.
\end{exmp}

\subsection{Primal-Dual Splitting}

If neither $\Prox_{\tau g \circ A}$ nor $\Prox_{\tau f^\star\circ A^*}$ is tractable, a primal-dual method can work directly with the saddle problem. Its efficiency depends on the available operator evaluations, proximal maps, and conditioning.
%
For the structured problem~\eqref{eq-splitting-primal-dual}, biconjugacy $g=(g^*)^*$ gives the saddle formulation
\begin{align}\label{eq-splitting-primal-dual-bis}
	\uinf{x} f(x) + g(Ax) = \uinf{x} f(x) + \usup{u} \dotp{Ax}{u} - g^*(u)
	= \usup{u} \uinf{x} f(x) + \dotp{Ax}{u}  - g^*(u).
\end{align}
We assume a primal-dual saddle point exists, so strong duality holds; strong convexity of $f$ is not required.

\begin{exmp}[Total variation regularization of inverse problems]
For example, TV regularization for the inverse problem $\Kk x=y$ leads to
\eq{
	\umin{x} \frac{1}{2}\norm{y-\Kk x}^2 + \la \norm{\nabla x}_{1,2},
}
where $A=\nabla$, $f(x)=\frac{1}{2}\norm{y-\Kk x}^2$, and $g=\la \norm{\cdot}_{1,2}$.
%
For this split, the proximal map of $f$ requires solving a system involving $\Id+\tau\Kk^*\Kk$, while the proximal map of $g^*$ is a pointwise projection. If that linear solve is costly, use instead $A x=(\Kk x,\nabla x)$, $f=0$, and $g(z,v)=\frac12\norm{y-z}^2+\la\norm{v}_{1,2}$. The two blocks of $g^*$ then have simple proximal maps, so each iteration uses only $\Kk,\Kk^*,\nabla$, and $\nabla^*$.
\end{exmp}

A standard primal-dual algorithm, described in~\cite{chambolle2016introduction}, initializes $\tilde x^0=x^0$ and updates
\begin{align*}
 \iit{z}&=\Prox_{\sigma g^*}\bigl(\it{z}+\sigma A\it{\tilde x}\bigr),\\
 \iit{x}&=\Prox_{\tau f}\bigl(\it{x}-\tau A^*\iit{z}\bigr),\\
 \iit{\tilde x}&=\iit{x}+\theta(\iit{x}-\it{x}).
\end{align*}
For $\theta=1$, $\sigma,\tau>0$, and $\sigma\tau\norm{A}^2<1$, the primal and dual iterates converge to a saddle point, provided one exists. The value $\theta=0$ is not covered by this general convergence guarantee.
