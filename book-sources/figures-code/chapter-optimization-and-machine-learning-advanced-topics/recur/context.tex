\subsection{Feed-forward Architecture}

A parameterized feedforward architecture, such as a neural network, has the form
\eql{\label{eq-feednets}
	\foralls k=1, \ldots, t, \quad x_{k} = f_k(x_{k-1},\th_{k-1})
}
where $\th_{k-1}$ is the parameter vector for a layer and the input $x_0 \in \RR^{n_0}$ is fixed. The objective is
\eql{\label{eq-loss-feedf}
	f(\th) \eqdef L(x_t)
}
Here $L : \RR^{n_t} \rightarrow \RR$ is a loss, such as squared prediction error or logistic loss, and $\th=(\th_k)_{k=0}^{t-1}$ collects the parameters. Figure~\ref{fig-mlp} shows the computational graph.

\caption{\label{fig-mlp}
Computational graph for a feedforward architecture.
}
\end{figure}


Reverse mode computes the gradient of $f$ by propagating derivatives through the variables $(x_k,\th_k)$. Initialize
\eq{
	\nabla_{x_t} f = \nabla L(x_t) 	
}
and recurse from $k=t$ down to $1$:
\eql{\label{eq-backprop-discr}
	z_{k-1} = [\partial_x f_k(x_{k-1},\th_{k-1})]^\top z_k
	\qandq
	\nabla_{\th_{k-1}} f = [\partial_\th f_k(x_{k-1},\th_{k-1})]^\top (\nabla_{x_{k}} f)
}
where $z_k \eqdef \nabla_{x_{k}} f(\th)$ is the gradient with respect to $x_{k}$.

   
\paragraph{Multilayer perceptrons.}\label{sec-autodiff-mlp}

For a fully connected network without biases, use
\eql{\label{eq-mlp-func}
	\foralls x_{k-1} \in \RR^{n_{k-1}}, \quad
	f_k(x_{k-1},\th_{k-1}) = \rho( \th_{k-1} x_{k-1} )
} 
where $\th_{k-1}\in\RR^{n_k\times n_{k-1}}$ contains the weights and $\rho$ acts coordinatewise; see Figure~\ref{fig-mlp-param}. The derivative formulas below apply where $\rho$ is differentiable. At a ReLU kink, a chosen derivative convention defines the computed update, but the classical derivative need not exist.
 
For an adjoint vector $z_k \in \RR^{n_k}$, typically $\nabla_{x_{k}} f$,
\eq{
	\left\{
	\begin{matrix}
	[\partial_x f_k(x_{k-1},\th_{k-1})]^\top (z_k) = \th_{k-1}^\top w_k z_k, \\
	[\partial_\th f_k(x_{k-1},\th_{k-1})]^\top (z_k) =(w_kz_k)x_{k-1}^\top
	\end{matrix}
	\right.
	\qwhereq
	w_k \eqdef \diag(\rho'(\th_{k-1} x_{k-1})).
}

\paragraph{Link with adjoint state method.}

A residual form of~\eqref{eq-feednets} can be interpreted as a time discretization of an ordinary differential equation.
 
Keep the state dimension fixed, $n_k=n$, and interpret $x_k$ as an approximation of a continuous trajectory $x(t)$ at time $t=k\tau$. The formal continuum limit takes $\tau\downarrow0$ with $k\tau$ approaching a fixed time.
 
Impose the residual structure
\eql{\label{eq-flow-eq}
	f_k(x_{k-1},\th_{k-1}) = x_{k-1} + \tau u(x_{k-1},\th_{k-1}, k\tau)
}
with a parameterized vector field $u(x,\th,t) \in \RR^n$. Formally, as $\tau \rightarrow 0$, this approaches the nonlinear ODE
\eql{\label{eq-ode-structure}
	\dot x(t) = u(x(t),\th(t),t)
}
with $x(t=0)=x_0$.

Writing $z(t) = \nabla_{x(t)} f(\th)$ for the adjoint, the discrete relations~\eqref{eq-backprop-discr} become a linear backward ODE:
\eq{
	\dot z(t)  =  - [\partial_x u(x(t),\th(t),t)]^\top z(t)
	\qandq
	\nabla_{\th(t)} f(\th) = [\partial_\th u(x(t),\th(t),t)]^\top z(t).
}
The adjoint has terminal condition $z(T)=\nabla L(x(T))$. The parameter derivative is interpreted as the $L^2$ functional gradient with respect to the time-dependent control. Its discrete normalization is $\frac{1}{\tau} \nabla_{\th_{k-1}} f  \rightarrow \nabla_{\th(t)} f(\th)$.

                                                   
\subsection{Recurrent Architectures}

A recurrent architecture reuses the parameter $\th=\th_k$ and map $f_k=h$ at every step of~\eqref{eq-feednets}:
\eql{\label{eq-feednets-recur}
	\foralls k=1, \ldots, t, \quad x_{k} = h(x_{k-1},\th).
}
Consider a scalar objective
\eq{
	f(\th) = L(x_t,\th)
}
The loss now depends explicitly on $\th$, extending~\eqref{eq-loss-feedf}. Figure~\ref{fig-recur} shows the corresponding graph.


\begin{figure}
\centering
\includegraphics[width=.6\linewidth]{auto-diff/archi-recurrent}
\caption{\label{fig-recur}
Computational graph for a recurrent architecture.
}
\end{figure}

Backpropagation gives
\eql{\label{eq-backprop-recurrent}
	\nabla_{x_{k-1}} f = [\partial_x h(x_{k-1},\th)]^\top \nabla_{x_{k}} f
	\qandq
	\nabla_{\th} f = 
		\nabla_\th L(x_t,\th) + 
		\sum_k [\partial_\th h(x_{k-1},\th)]^\top \nabla_{x_{k}} f.
}
Similarly, with $h(x,\th) = x + \tau u(x,\th)$, the formal limit $(k,k\tau) \rightarrow (+\infty,t)$ gives a forward nonlinear ODE with a time-independent vector field:
\eq{
	\dot x(t) = u(x(t),\th)
}
and the following linear backward adjoint equation, for $f(\th)=L(x(T),\th)$
\eql{\label{eq-jacobian-mlp}
	\dot z(t)  =  - [\partial_x u(x(t),\th)]^\top z(t)
	\qandq
	\nabla_{\th} f(\th) = \nabla_\th L(x(T),\th) + \int_0^T[\partial_\th u(x(t),\th)]^\top z(t) \d t.
}
with terminal condition $z(T)=\nabla_xL(x(T),\th)$.

     
\paragraph{Residual recurrent networks. }

A residual recurrent network uses the update
\eq{
	h(x,\th) = x + W_2^\top \rho( W_1 x )
}
as illustrated in Figure~\ref{fig-recurrent-param}. Here $\th = (W_1, W_2) \in (\RR^{q\times n})^2$ contains the weights and $\rho$ is a pointwise activation. Increasing the hidden width $q$ enlarges the class of representable residual maps.
 
When $W_2=-\tau W_1$ and $\rho=\psi'$, this update is a gradient step for $\Ee(x,\th)=\sum_i\psi((W_1x)_i)$. Repeating it gives an iterative implementation of the argmin layer~\eqref{eq-argmin-layer}, provided the iterations converge to a minimizer.
 
The Jacobians $\partial_\th h$ and $\partial_x h$ follow from the layer formulas in the \hyperref[sec-autodiff-mlp]{multilayer-perceptron discussion}.
