% Landscape review pages keep the original chapter figures and labels intact.
\usepackage{pdflscape}
\newcommand{\BookFigureReview}[1]{%
  \IfFileExists{figures/tikz/reviews/#1.tex}{%
    \clearpage\begingroup
    \pagestyle{empty}\setlength{\parindent}{0pt}%
    \begin{landscape}
      \phantomsection
      \addcontentsline{toc}{section}{Figure reconstructions for review}%
      \input{figures/tikz/reviews/#1.tex}%
    \end{landscape}\endgroup
  }{}%
}
